import type { Request, Response } from "express";
import { AuthService } from "../Service/authService";
import { RevocationService } from "../Service/revocationService";

const isProduction =
  process.env.NODE_ENV === "production" || process.env.VERCEL === "1";

// Cấu hình chung cho Cookie chứa Refresh Token
const REFRESH_COOKIE_OPTS = {
  httpOnly: true,
  secure: isProduction, // Bắt buộc true khi chạy HTTPS (Vercel)
  sameSite: isProduction ? ("none" as const) : ("lax" as const), // 'none' an toàn nhất khi có thể có domain phụ
  maxAge: 7 * 24 * 60 * 60 * 1000,
  path: "/api/auth",
};

export class AuthController {
  /**
   * Đăng nhập: Gửi Refresh Token qua HttpOnly Cookie, Access Token qua JSON
   */
  static async login(req: Request, res: Response) {
    try {
      const result = await AuthService.login(req.body);
      const { refreshToken, accessToken, user } = result;

      // Set Refresh Token vào Cookie
      res.cookie("refreshToken", refreshToken, REFRESH_COOKIE_OPTS);

      return res.json({
        success: true,
        message: "Đăng nhập thành công!",
        data: {
          user,
          accessToken,
        },
      });
    } catch (err: any) {
      return res.status(err.statusCode || 400).json({
        success: false,
        message: err.message || "Đăng nhập thất bại",
      });
    }
  }

  /**
   * Cấp lại Access Token mới dựa vào Refresh Token trong Cookie
   */
  static async refreshToken(req: Request, res: Response) {
    try {
      const refreshToken = req.cookies?.refreshToken;

      if (!refreshToken) {
        return res.status(401).json({
          success: false,
          message: "Không tìm thấy phiên làm việc! Vui lòng đăng nhập lại.",
        });
      }

      // Service trả về { accessToken, refreshToken, user }
      const result = await AuthService.refreshToken(refreshToken);

      // Ghi đè Refresh Token mới vào Cookie (bắt buộc do cơ chế rotation)
      res.cookie("refreshToken", result.refreshToken, REFRESH_COOKIE_OPTS);

      return res.status(200).json({
        success: true,
        data: {
          accessToken: result.accessToken,
          user: result.user,
        },
      });
    } catch (err: any) {
      // Xóa cookie nếu refresh token không hợp lệ để buộc đăng nhập lại
      res.clearCookie("refreshToken", { ...REFRESH_COOKIE_OPTS, maxAge: 0 });

      return res.status(err.statusCode || 401).json({
        success: false,
        message: err.message || "Phiên làm việc đã hết hạn!",
      });
    }
  }

  /**
   * Đăng xuất: Thu hồi token trên server và Xóa Cookie
   */
  static async logout(req: Request, res: Response) {
    try {
      const accessToken = req.headers.authorization?.split(" ")[1];
      const refreshToken = req.cookies?.refreshToken;

      // Đưa cả Access và Refresh Token vào danh sách đen
      await RevocationService.revokePair(accessToken, refreshToken);

      // Xóa cookie
      res.clearCookie("refreshToken", { ...REFRESH_COOKIE_OPTS, maxAge: 0 });

      return res.json({
        success: true,
        message: "Đăng xuất thành công!",
      });
    } catch (err: any) {
      return res.status(500).json({
        success: false,
        message: err.message || "Lỗi khi đăng xuất",
      });
    }
  }

  /**
   * Đăng ký thông thường
   */
  static async register(req: Request, res: Response) {
    try {
      const newUser = await AuthService.register(req.body);
      return res.status(201).json({
        success: true,
        message: "Đăng ký thành công!",
        data: newUser,
      });
    } catch (err: any) {
      return res.status(err.statusCode || 400).json({ message: err.message });
    }
  }

  /**
   * Yêu cầu gửi OTP
   */
  static async requestOtp(req: Request, res: Response) {
    try {
      const { email, type } = req.body;
      await AuthService.requestOtp(email, type);
      return res.json({
        success: true,
        message: "Mã OTP đã được gửi đến email của bạn.",
      });
    } catch (err: any) {
      return res.status(err.statusCode || 400).json({ message: err.message });
    }
  }

  /**
   * Đăng ký hoàn tất kèm OTP
   */
  static async completeRegister(req: Request, res: Response) {
    try {
      const newUser = await AuthService.completeRegister(req.body);
      return res.status(201).json({
        success: true,
        message: "Đăng ký tài khoản thành công!",
        userId: newUser.id,
      });
    } catch (err: any) {
      return res.status(err.statusCode || 400).json({ message: err.message });
    }
  }

  /**
   * Đặt lại mật khẩu bằng OTP
   */
  static async resetPassword(req: Request, res: Response) {
    try {
      await AuthService.resetPassword(req.body);
      return res.json({
        success: true,
        message: "Đặt lại mật khẩu thành công! Vui lòng đăng nhập lại.",
      });
    } catch (err: any) {
      return res.status(err.statusCode || 400).json({ message: err.message });
    }
  }
}

import type { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
import { JWT_SECRET } from "../config/jwt";
import { RevocationService } from "../Service/revocationService";

export const authenticateToken = async (
  req: Request,
  res: Response,
  next: NextFunction,
) => {
  const authHeader = req.headers.authorization;
  const token = authHeader && authHeader.split(" ")[1];

  if (!token) {
    return res.status(401).json({
      success: false,
      message: "Access Token không được cung cấp!",
    });
  }

  try {
    // Dùng verify đồng bộ (không truyền callback) để dễ dùng async/await
    const decoded = jwt.verify(token, JWT_SECRET) as any;

    //  Kiểm tra đúng loại token (chặn dùng Refresh Token gọi API)
    if (decoded.type !== "access") {
      return res.status(401).json({
        success: false,
        message: "Sai loại token. Vui lòng sử dụng Access Token!",
      });
    }

    // Kiểm tra xem token đã bị thu hồi chưa (khi user logout)
    const isRevoked = await RevocationService.isRevoked(decoded.jti);
    if (isRevoked) {
      return res.status(401).json({
        success: false,
        message: "Phiên đăng nhập đã kết thúc hoặc token bị thu hồi!",
      });
    }

    // 3. Gắn thông tin user vào request để các controller phía sau dùng
    (req as any).user = decoded;
    next();
  } catch (err: any) {
    console.log("❌ LỖI VERIFY JWT:", err.message);
    return res.status(401).json({
      success: false,
      message: "Token không hợp lệ hoặc đã hết hạn!",
    });
  }
};

/**
 * Helper lấy Redis Client đang hoạt động (Primary -> Backup)
 */
const getActiveRedis = (): Redis | null => {
  if (redisPrimary.status === "ready" || redisPrimary.status === "connect") {
    return redisPrimary;
  }
  if (
    redisBackup &&
    (redisBackup.status === "ready" || redisBackup.status === "connect")
  ) {
    return redisBackup;
  }
  return null;
};

export const idempotency = (ttlSeconds: number = 86400) => {
  return async (
    req: Request,
    res: Response,
    next: NextFunction,
  ): Promise<void> => {
    const idempotencyKey = req.headers["x-idempotency-key"] as string;

    // Nếu không có header idempotency key, tiếp tục luồng bình thường
    if (!idempotencyKey) {
      return next();
    }

    const cacheKey = `idempotency:${idempotencyKey}`;
    const activeRedis = getActiveRedis();

    // 1. Kiểm tra cache nếu Redis khả dụng
    if (activeRedis) {
      try {
        const cachedResponse = await activeRedis.get(cacheKey);

        // Nếu request đã được xử lý trước đó -> Trả về kết quả cũ
        if (cachedResponse) {
          const { statusCode, body } = JSON.parse(cachedResponse);
          res.status(statusCode).json(body);
          return;
        }
      } catch (error: any) {
        console.warn(
          ` [IDEMPOTENCY WARNING] Lỗi đọc Redis: ${error.message}`,
        );
      }
    }

    // 2. Ghi đè res.json để tự động cache response sau khi controller xử lý xong
    const originalJson = res.json.bind(res);
    res.json = (body: any) => {
      if (res.statusCode >= 200 && res.statusCode < 300) {
        const currentRedis = getActiveRedis();
        if (currentRedis) {
          currentRedis
            .setex(
              cacheKey,
              ttlSeconds,
              JSON.stringify({ statusCode: res.statusCode, body }),
            )
            .catch((err) =>
              console.warn(
                ` [IDEMPOTENCY WARNING] Lỗi ghi Redis: ${err.message}`,
              ),
            );
        }
      }
      return originalJson(body);
    };

    next();
  };
};

import type { Request, Response, NextFunction } from "express";

export const checkIpBlacklist = async (
  req: Request,
  res: Response,
  next: NextFunction,
) => {
  try {
    // Lấy IP thực (hỗ trợ sau Proxy/Nginx/Cloudflare)
    const clientIp =
      (req.headers["x-forwarded-for"] as string)?.split(",")[0]?.trim() ||
      (req.headers["cf-connecting-ip"] as string) ||
      req.socket.remoteAddress ||
      "127.0.0.1";

    const cacheKey = `ipbl:${clientIp}`;
    let cachedStatus: string | null = null;

    // 1. Check Cache trên Redis trước (giảm tải DB)
    try {
      cachedStatus = await redisPrimary.get(cacheKey);
    } catch (redisErr) {
      console.warn("[IP GUARD] Redis lỗi, fallback xuống DB:", redisErr);
    }

    // Nếu cache trả về "1" (đang bị block)
    if (cachedStatus === "1") {
      return res.status(403).json({
        success: false,
        code: "IP_BLOCKED",
        message: `IP của bạn (${clientIp}) bị hạn chế đăng ký tài khoản mới do liên quan đến khoản nợ/vi phạm chưa xử lý.`,
      });
    }

    // Nếu cache trả về "0" (sạch), cho qua luôn
    if (cachedStatus === "0") {
      return next();
    }

    // 2. Nếu Cache không có (null), mới query DB
    const blacklisted = await prisma.blacklistedIPs.findFirst({
      where: {
        ipAddress: clientIp,
        isResolved: false,
      },
    });

    // Lưu lại kết quả vào Redis trong 30 giây
    try {
      await redisPrimary.set(cacheKey, blacklisted ? "1" : "0", "EX", 30);
    } catch (redisErr) {
      console.warn("[IP GUARD] Không lưu được cache IP:", redisErr);
    }

    if (blacklisted) {
      return res.status(403).json({
        success: false,
        code: "IP_BLOCKED",
        message: `IP của bạn (${clientIp}) bị hạn chế đăng ký tài khoản mới do liên quan đến khoản nợ/vi phạm chưa xử lý.`,
      });
    }

    next();
  } catch (error) {
    // Lỗi DB thì vẫn cho đi tiếp để không gián đoạn hệ thống (Fail-open)
    console.error("[IP GUARD] Lỗi kiểm tra IP:", error);
    next();
  }
};

import type { Request, Response, NextFunction } from "express";
import { ZodError, type ZodTypeAny } from "zod";

export class AppError extends Error {
  statusCode: number;
  details?: any;

  constructor(message: string, statusCode: number = 400, details?: any) {
    super(message);
    this.statusCode = statusCode;
    this.details = details;
  }
}

export const validate = (schema: ZodTypeAny) => {
  return async (
    req: Request,
    res: Response,
    next: NextFunction,
  ): Promise<void> => {
    try {
      // 1. Parse & Validate và cast thành `any` để truy cập các thuộc tính body, query, params
      const parsed = (await schema.parseAsync({
        body: req.body,
        query: req.query,
        params: req.params,
      })) as any;

      // 2. Gán lại dữ liệu đã qua sanitize / transform / coerce
      if (parsed.body !== undefined) {
        req.body = parsed.body;
      }

      if (parsed.query !== undefined) {
        for (const key in req.query) delete req.query[key];
        Object.assign(req.query, parsed.query);
      }

      if (parsed.params !== undefined) {
        for (const key in req.params) delete req.params[key];
        Object.assign(req.params, parsed.params);
      }

      next();
    } catch (error: any) {
      if (error instanceof ZodError) {
        const issueDetails = error.issues.map((issue) => ({
          field: issue.path.join("."),
          message: issue.message,
        }));

        const errorMessage = issueDetails.map((i) => i.message).join("; ");

        // Đẩy lỗi chuẩn về Global Error Handler
        return next(new AppError(errorMessage, 400, issueDetails));
      }

      next(error);
    }
  };
};

    "Thiếu JWT_ACCESS_SECRET hoặc JWT_REFRESH_SECRET trong .env!",
  );
}

export class AuthService {
  /**
   * Hàm dùng chung để tạo cặp Access Token & Refresh Token
   */
  static generateTokensForUser(user: any) {
    const accessPayload = {
      id: user.id,
      userId: user.id,
      email: user.email,
      role: user.role,
      type: "access" as const,
    };

    const refreshPayload = {
      id: user.id,
      userId: user.id,
      type: "refresh" as const,
    };

    const accessToken = jwt.sign(accessPayload, ACCESS_TOKEN_SECRET!, {
      expiresIn: "15m",
      jwtid: crypto.randomUUID(),
    });

    const refreshToken = jwt.sign(refreshPayload, REFRESH_TOKEN_SECRET!, {
      expiresIn: "7d",
      jwtid: crypto.randomUUID(),
    });

    return { accessToken, refreshToken };
  }

  /**
   * Đăng nhập hệ thống
   */
  static async login(data: {
    email: string;
    pass?: string;
    password?: string;
  }) {
    const inputPassword = data.pass || data.password || "";

    const user = await prisma.users.findUnique({
      where: { email: data.email.toLowerCase().trim() },
      include: { wallet: true },
    });

    if (!user) {
      const error: any = new Error("Email hoặc mật khẩu không chính xác!");
      error.statusCode = 400;
      throw error;
    }

    // BẢO MẬT: Nếu user chưa thiết lập mật khẩu (chỉ dùng Passkey)
    if (!user.password) {
      const error: any = new Error(
        "Tài khoản này chưa thiết lập mật khẩu. Vui lòng đăng nhập bằng Vân tay (Passkey)!",
      );
      error.statusCode = 400;
      throw error;
    }

    const isPasswordValid = await bcrypt.compare(inputPassword, user.password);
    if (!isPasswordValid) {
      const error: any = new Error("Email hoặc mật khẩu không chính xác!");
      error.statusCode = 400;
      throw error;
    }

    const tokens = this.generateTokensForUser(user);

    // Loại bỏ mật khẩu và dữ liệu nhạy cảm trước khi trả về
    const { password, currentChallenge, ...userWithoutPassword } = user;

    return {
      ...tokens,
      user: userWithoutPassword,
    };
  }

  /**
   * Cấp lại Access Token mới dựa vào Refresh Token (có ROTATION)
   */
  static async refreshToken(refreshToken: string) {
    try {
      const decoded = jwt.verify(refreshToken, REFRESH_TOKEN_SECRET!) as {
        id: number;
        userId?: number;
        type?: string;
        jti?: string;
        exp?: number;
      };

      if (decoded.type !== "refresh") {
        throw new Error("Sai loại token!");
      }

      if (decoded.jti && (await RevocationService.isRevoked(decoded.jti))) {
        throw new Error("Refresh token đã bị thu hồi!");
      }

      const user = await prisma.users.findUnique({
        where: { id: decoded.id ?? decoded.userId! },
        include: { wallet: true },
      });

      if (!user) {
        const error: any = new Error("Người dùng không tồn tại!");
        error.statusCode = 401;
        throw error;
      }

      if (decoded.jti && decoded.exp) {
        await RevocationService.revoke(decoded.jti, decoded.exp);
      }

      const tokens = this.generateTokensForUser(user);
      const { password, currentChallenge, ...userWithoutPassword } = user;

      return {
        accessToken: tokens.accessToken,
        refreshToken: tokens.refreshToken,
        user: userWithoutPassword,
      };
    } catch (err: any) {
      if (err.statusCode) throw err;
      const error: any = new Error(
        "Phiên đăng nhập đã hết hạn. Vui lòng đăng nhập lại!",
      );
      error.statusCode = 401;
      throw error;
    }
  }

  /**

    const encryptedPhone = phoneNumber

      },
      select: {

  static async requestOtp(email: string, type: "REGISTER" | "FORGOT_PASSWORD") {

import { LRUCache } from "lru-cache";
import { redisPrimary, redisBackup } from "../Utils/redisLock";
import type Redis from "ioredis";

// L1: cache RAM local, tối đa 500 key, TTL 10s
const localCache = new LRUCache<string, any>({
  max: 500,
  ttl: 1000 * 10,
});

// Chuyển pattern kiểu Redis (`trip:*`) sang regex để lọc L1
function patternToRegex(pattern: string): RegExp {
  const escaped = pattern
    .replace(/[.+?^${}()|[\]\\]/g, "\\$&")
    .replace(/\*/g, ".*");
  return new RegExp(`^${escaped}$`);
}

export class MultiLevelCache {
  /**
   * Lấy Redis đang sống (Primary → Backup).
   */
  private static getActiveRedis(): Redis | null {
    if (redisPrimary.status === "ready" || redisPrimary.status === "connect") {
      return redisPrimary;
    }
    if (
      redisBackup &&
      (redisBackup.status === "ready" || redisBackup.status === "connect")
    ) {
      console.warn("[CACHE FAILOVER] Đang dùng Redis Backup cho L2!");
      return redisBackup;
    }
    return null;
  }

  /**
   * Đọc dữ liệu: L1 (RAM) → L2 (Redis) → null (để caller tự query DB).
   * Có log thời gian từng tầng để đo hiệu quả cache.
   */
  static async get<T>(key: string): Promise<T | null> {
    const t0 = process.hrtime.bigint();

    // 1. L1 (RAM)
    if (localCache.has(key)) {
      const ms = Number(process.hrtime.bigint() - t0) / 1e6;
      console.log(`[CACHE L1] ${key} = ${ms.toFixed(2)}ms`);
      return localCache.get(key) as T;
    }

    // 2. L2 (Redis)
    const activeRedis = this.getActiveRedis();
    if (activeRedis) {
      try {
        const redisData = await activeRedis.get(key);
        if (redisData) {
          const parsed = JSON.parse(redisData);
          localCache.set(key, parsed); // đồng bộ ngược lại L1
          const ms = Number(process.hrtime.bigint() - t0) / 1e6;
          console.log(`[CACHE L2] ${key} = ${ms.toFixed(2)}ms`);
          return parsed;
        }
      } catch (error: any) {
        console.warn(`[CACHE WARNING] Lỗi đọc Redis: ${error.message}`);
      }
    }

    const ms = Number(process.hrtime.bigint() - t0) / 1e6;
    console.log(`[CACHE MISS] ${key} = ${ms.toFixed(2)}ms`);
    return null;
  }

  /**
   * Ghi dữ liệu đồng thời vào L1 (RAM) và L2 (Redis).
   */
  static async set(
    key: string,
    value: any,
    ttlSeconds: number = 300,
  ): Promise<void> {
    localCache.set(key, value, { ttl: ttlSeconds * 1000 });

    const activeRedis = this.getActiveRedis();
    if (activeRedis) {
      try {
        await activeRedis.setex(key, ttlSeconds, JSON.stringify(value));
      } catch (error: any) {
        console.warn(`[CACHE WARNING] Lỗi ghi Redis: ${error.message}`);
      }
    }
  }

  /**
   * Xóa 1 key cụ thể ở cả L1 và L2.
   */
  static async delete(key: string): Promise<void> {
    localCache.delete(key);
    const activeRedis = this.getActiveRedis();
    if (activeRedis) {
      try {
        await activeRedis.del(key);
      } catch (error: any) {
        console.warn(`[CACHE WARNING] Lỗi xóa Redis: ${error.message}`);
      }
    }
  }

  /**
   * Xóa cache theo pattern (VD: "trip:123*").
   * Dùng SCAN thay KEYS để không block Redis.
   * Chỉ xóa các key L1 khớp pattern, không xóa toàn bộ.
   */
  static async invalidate(keyPattern: string): Promise<void> {
    // 1. Xóa L1 theo pattern (không clear tất cả)
    const regex = patternToRegex(keyPattern);
    for (const key of localCache.keys()) {
      if (regex.test(key)) localCache.delete(key);
    }

    // 2. Xóa L2 bằng SCAN (an toàn với Redis lớn)
    const activeRedis = this.getActiveRedis();
    if (!activeRedis) return;

    try {
      let cursor = "0";
      let deletedCount = 0;
      do {
        const [nextCursor, keys] = await activeRedis.scan(
          cursor,
          "MATCH",
          keyPattern,
          "COUNT",
          100,
        );
        cursor = nextCursor;
        if (keys.length > 0) {
          await activeRedis.del(...keys);
          deletedCount += keys.length;
        }
      } while (cursor !== "0");

      if (deletedCount > 0) {
        console.log(
          `[CACHE INVALIDATE] ${keyPattern} → xóa ${deletedCount} key L2`,
        );
      }
    } catch (error: any) {
      console.warn(
        `[CACHE WARNING] Lỗi invalidate "${keyPattern}": ${error.message}`,
      );
    }
  }
}

import { v2 as cloudinary } from "cloudinary"; // Dùng singleton chung
import {
  cloudinaryPrimaryConfig,
  cloudinaryBackupConfig,
} from "../config/cloudinary";
import { Readable } from "stream";
import prisma from "../database/prismaClient";

export class CloudinaryService {
  /**
   * Helper: Upload lên 1 bucket cụ thể bằng cách tiêm (inject) config vào options
   * Việc này giúp tránh hoàn toàn lỗi ghi đè singleton của thư viện Cloudinary.
   */
  private static uploadToBucket(
    config: typeof cloudinaryPrimaryConfig,
    fileBuffer: Buffer,
    userId: number,
  ): Promise<string> {
    return new Promise((resolve, reject) => {
      const uploadStream = cloudinary.uploader.upload_stream(
        {
          ...config, // Bơm credentials (cloud_name, api_key...) trực tiếp vào đây
          folder: "nexus_ticket/avatars",
          public_id: `user_${userId}`,
          overwrite: true,
          transformation: [
            { width: 250, height: 250, crop: "fill", gravity: "face" },
            { quality: "auto", fetch_format: "auto" },
          ],
        },
        (error, result) => {
          if (error || !result) {
            return reject(error || new Error("Lỗi upload ảnh lên Cloudinary!"));
          }
          resolve(result.secure_url);
        },
      );

      Readable.from(fileBuffer).pipe(uploadStream);
    });
  }

  /**
   * Upload Avatar Stream với cơ chế Retry (Primary) và Auto-Failover (Backup)
   */
  static async uploadAvatarStream(
    fileBuffer: Buffer,
    userId: number,
  ): Promise<string> {
    let lastPrimaryError: any;

    // 1. Thử Upload qua Cloudinary Primary (Thử tối đa 3 lần)
    for (let attempt = 0; attempt < 3; attempt++) {
      try {
        return await this.uploadToBucket(
          cloudinaryPrimaryConfig,
          fileBuffer,
          userId,
        );
      } catch (e: any) {
        lastPrimaryError = e;
        console.warn(
          `[CLOUDINARY] Primary lỗi lần ${attempt + 1}/3: ${e.message}`,
        );
        // Chờ 0.5s -> 1s -> 2s trước khi thử lại
        await new Promise((r) => setTimeout(r, 2 ** attempt * 500));
      }
    }

    // 2. Fallback sang Cloudinary Backup
    console.warn(
      `[CLOUDINARY WARNING] Primary sập hoàn toàn. Chuyển sang Backup...`,
    );
    try {
      return await this.uploadToBucket(
        cloudinaryBackupConfig,
        fileBuffer,
        userId,
      );
    } catch (backupError: any) {
      console.error(
        "[CLOUDINARY CRITICAL] Cả 2 tài khoản Cloudinary đều thất bại!",
      );
      throw new Error(
        `Không thể upload avatar: Primary=${lastPrimaryError?.message}, Backup=${backupError.message}`,
      );
    }
  }

  /**
   * Cập nhật Avatar vào DB và tự động dọn dẹp ảnh cũ trên Cloudinary
   */
  static async updateAvatar(userId: number, fileBuffer: Buffer) {
    // Lấy URL ảnh cũ trước khi cập nhật
    const oldUser = await prisma.users.findUnique({
      where: { id: userId },
      select: { avatarUrl: true },
    });

    // Upload ảnh mới (Có retry + failover)
    const newAvatarUrl = await this.uploadAvatarStream(fileBuffer, userId);

    // Lưu link mới vào DB
    const updatedUser = await prisma.users.update({
      where: { id: userId },
      data: { avatarUrl: newAvatarUrl },
      select: {
        id: true,
        email: true,
        fullName: true,
        role: true,
        avatarUrl: true,
      },
    });

    // Xóa ảnh cũ khỏi Cloudinary để đỡ tốn dung lượng
    if (oldUser?.avatarUrl && oldUser.avatarUrl !== newAvatarUrl) {
      try {
        const match = oldUser.avatarUrl.match(/\/v\d+\/(.+)\.[a-z]+$/i);
        if (match && match[1]) {
          // Lưu lại config hiện tại
          const currentConfig = cloudinary.config();

          // Chuyển tạm sang config Primary để xóa cho đúng bucket
          cloudinary.config(cloudinaryPrimaryConfig);

          // Hàm destroy tự động dùng config global, không nhận config trong options
          await cloudinary.uploader.destroy(match[1]);

          // Khôi phục lại config cũ
          cloudinary.config(currentConfig);

          console.log(`[CLOUDINARY] Đã dọn dẹp ảnh cũ của user ${userId}`);
        }
      } catch (err: any) {
        // Chỉ log lỗi, không throw để tránh làm hỏng luồng update ảnh mới
        console.error("[CLOUDINARY] Không xóa được ảnh cũ:", err.message);
      }
    }

    return updatedUser;
  }
}

import prisma from "../database/prismaClient";
import { mailQueue } from "../queues/mailQueue";
import crypto from "crypto";

// Số lần nhập sai tối đa trước khi vô hiệu OTP
const MAX_ATTEMPTS = 5;

export class OtpService {
  /**
   * Tạo OTP, lưu DB, đẩy job gửi email vào BullMQ.
   * Cooldown 60s chống spam + xóa OTP cũ của cùng email/type.
   */
  static async createOTPLogic(
    email: string,
    type: "REGISTER" | "FORGOT_PASSWORD" = "REGISTER",
  ) {
    const cleanEmail = email.toLowerCase().trim();
    const now = new Date();

    // Chặn spam: không cho yêu cầu mã mới trong vòng 1 phút
    const oneMinuteAgo = new Date(now.getTime() - 1 * 60 * 1000);
    const standardSpamCheck = await prisma.oTP.findFirst({
      where: {
        email: cleanEmail,
        type: type as any,
        createAt: { gte: oneMinuteAgo },
      },
    });

    if (standardSpamCheck) {
      const error: any = new Error(
        "Vui lòng đợi 1 phút trước khi yêu cầu mã mới!",
      );
      error.statusCode = 429;
      throw error;
    }

    const otpCode = crypto.randomInt(100000, 999999).toString();
    const expiredAt = new Date(now.getTime() + 5 * 60 * 1000);

    // Xóa OTP cũ của email + type (chỉ giữ 1 mã hiệu lực)
    await prisma.oTP.deleteMany({
      where: { email: cleanEmail, type: type as any },
    });

    // ✅ attempts khởi tạo 0, tăng mỗi lần nhập sai
    const newOtp = await prisma.oTP.create({
      data: {
        email: cleanEmail,
        code: otpCode,
        type: type as any,
        expiredAt,
        attempts: 0,
      },
    });

    try {
      await mailQueue.add(
        "send-otp-job",
        { email: cleanEmail, code: otpCode, type },
        {
          attempts: 3,
          backoff: { type: "exponential", delay: 2000 },
          removeOnComplete: true,
          removeOnFail: 100,
        },
      );
      console.log(`[QUEUE] Đã đẩy job gửi OTP của ${cleanEmail} vào BullMQ`);
    } catch (queueErr: any) {
      // Queue lỗi → xóa OTP vừa tạo để user không bị kẹt cooldown
      await prisma.oTP.delete({ where: { id: newOtp.id } }).catch(() => {});
      console.error("[QUEUE] Đẩy job thất bại:", queueErr.message);
      const error: any = new Error(
        "Không gửi được mã OTP, vui lòng thử lại sau ít phút!",
      );
      error.statusCode = 503;
      throw error;
    }

    // Chỉ trả field an toàn, KHÔNG trả `code` cho client
    return {
      email: newOtp.email,
      type: newOtp.type,
      expiredAt: newOtp.expiredAt,
    };
  }

  /**
   * Verify OTP với giới hạn 5 lần nhập sai (dùng DB attempts).
   * Sai quá 5 lần → xóa OTP, yêu cầu mã mới.
   */
  static async verifyOTPLogic(
    email: string,
    code: string,
    type: "REGISTER" | "FORGOT_PASSWORD" = "REGISTER",
  ): Promise<boolean> {
    const cleanEmail = email.toLowerCase().trim();
    const cleanCode = code.trim();

    // Lấy OTP mới nhất của email + type (bất kể code đúng/sai)
    const otpRecord = await prisma.oTP.findFirst({
      where: { email: cleanEmail, type: type as any },
      orderBy: { createAt: "desc" },
    });

    if (!otpRecord) {
      const error: any = new Error("Mã OTP không chính xác hoặc đã hết hạn!");
      error.statusCode = 400;
      throw error;
    }

    // Chặn brute-force: đã sai quá 5 lần
    if (otpRecord.attempts >= MAX_ATTEMPTS) {
      await prisma.oTP.deleteMany({
        where: { email: cleanEmail, type: type as any },
      });
      const error: any = new Error(
        `Bạn đã nhập sai quá ${MAX_ATTEMPTS} lần. Vui lòng yêu cầu mã OTP mới!`,
      );
      error.statusCode = 429;
      throw error;
    }

    // Kiểm tra hết hạn
    if (new Date() > otpRecord.expiredAt) {
      const error: any = new Error("Mã OTP đã hết hạn!");
      error.statusCode = 400;
      throw error;
    }

    // Mã sai → tăng attempts
    if (otpRecord.code !== cleanCode) {
      await prisma.oTP.update({
        where: { id: otpRecord.id },
        data: { attempts: { increment: 1 } },
      });
      const remaining = MAX_ATTEMPTS - otpRecord.attempts - 1;
      const error: any = new Error(

import crypto from "crypto";

const ENCRYPTION_KEY = process.env.ENCRYPTION_KEY;
if (!ENCRYPTION_KEY || ENCRYPTION_KEY.length !== 64) {
  throw new Error(
    "Thiếu hoặc sai định dạng ENCRYPTION_KEY (phải là 64 ký tự Hex)!",
  );
}

const ALGORITHM = "aes-256-gcm";
const keyBuffer = Buffer.from(ENCRYPTION_KEY, "hex");

export class EncryptionService {
  static encrypt(text: string): string {
    if (!text) return text;
    const iv = crypto.randomBytes(12);
    const cipher = crypto.createCipheriv(ALGORITHM, keyBuffer, iv);

    let encrypted = cipher.update(text, "utf8", "hex");
    encrypted += cipher.final("hex");
    const authTag = cipher.getAuthTag().toString("hex");

    return `${iv.toString("hex")}:${authTag}:${encrypted}`;
  }

  static decrypt(encryptedData: string): string {
    if (!encryptedData || !encryptedData.includes(":")) return encryptedData;

    const [ivHex, authTagHex, encryptedText] = encryptedData.split(":");
    const decipher = crypto.createDecipheriv(
      ALGORITHM,
      keyBuffer,
      Buffer.from(ivHex, "hex"),
    );

    decipher.setAuthTag(Buffer.from(authTagHex, "hex"));
    let decrypted = decipher.update(encryptedText, "hex", "utf8");
    decrypted += decipher.final("utf8");

    return decrypted;
  }
}

import axios from "axios";
import crypto from "crypto";

export class PayOS {
  private clientId: string;
  private apiKey: string;
  private checksumKey: string;

  constructor() {
    this.clientId = process.env.PAYOS_CLIENT_ID!;
    this.apiKey = process.env.PAYOS_API_KEY!;
    this.checksumKey = process.env.PAYOS_CHECKSUM_KEY!;
  }

  private sortObjDataByKey(object: any) {
    return Object.keys(object)
      .sort()
      .reduce((obj: any, key) => {
        obj[key] = object[key];
        return obj;
      }, {});
  }

  private convertObjToQueryStr(object: any) {
    return Object.keys(object)
      .filter((key) => object[key] !== undefined)
      .map((key) => {
        let value = object[key];
        if (value && Array.isArray(value)) {
          value = JSON.stringify(
            value.map((val) => this.sortObjDataByKey(val)),
          );
        }
        if ([null, undefined, "undefined", "null"].includes(value)) {
          value = "";
        }
        return `${key}=${value}`;
      })
      .join("&");
  }

  public createSignature(data: any) {
    const sortedData = this.sortObjDataByKey(data);
    const dataQueryStr = this.convertObjToQueryStr(sortedData);
    return crypto
      .createHmac("sha256", this.checksumKey)
      .update(dataQueryStr)
      .digest("hex");
  }

  public async createPaymentLink(data: any) {
    const signData = {
      amount: data.amount,
      cancelUrl: data.cancelUrl,
      description: data.description,
      orderCode: data.orderCode,
      returnUrl: data.returnUrl,
    };
    const signature = this.createSignature(signData);

    try {
      const res = await axios.post(
        "https://api-merchant.payos.vn/v2/payment-requests",
        { ...data, signature },
        {
          headers: {
            "x-client-id": this.clientId,
            "x-api-key": this.apiKey,
          },
        },
      );
      if (res.data.code === "00") return res.data.data;
      throw new Error(res.data.desc);
    } catch (err: any) {
      throw new Error(err.response?.data?.desc || err.message);
    }
  }

  public verifyWebhookData(webhookBody: any) {
    const { data, signature } = webhookBody;
    if (!data || !signature) return false;
    const mySignature = this.createSignature(data);
    return mySignature === signature;
  }
}

export const payosInstance = new PayOS();

import Redis from "ioredis";
import crypto from "crypto";
import dotenv from "dotenv";

dotenv.config();

const primaryUrl = process.env.REDIS_URL;
const backupUrl = process.env.REDIS_URL_BACKUP;

if (!primaryUrl) {
  throw new Error("Khai báo thiếu REDIS_URL trong file .env!");
}

const commonRedisOptions = {
  tls: {
    rejectUnauthorized: false,
  },
  family: 4,
  maxRetriesPerRequest: null,
  enableReadyCheck: false,
  keepAlive: 10000,
  retryStrategy(times: number) {

    return delay;
  },

// 1. Khởi tạo Primary Redis
export const redisPrimary = new Redis(primaryUrl, commonRedisOptions);

// 2. Khởi tạo Backup Redis
export const redisBackup = backupUrl
  ? new Redis(backupUrl, commonRedisOptions)
  : null;

redisPrimary.on("connect", () => {
  console.log("[REDIS PRIMARY]: Kết nối Redis Cloud thành công!");
});

redisPrimary.on("error", (err) => {
  console.error("[REDIS PRIMARY WARNING]:", err.message);
});

if (redisBackup) {
  redisBackup.on("connect", () => {
    console.log("[REDIS BACKUP]: Kết nối Redis Cloud Dự Phòng sẵn sàng!");
  });
  redisBackup.on("error", (err) => {
    console.error("[REDIS BACKUP WARNING]:", err.message);
  });

// Script Lua: chỉ set nếu chưa có key (atomic NX)
const ACQUIRE_LUA = `
  if redis.call("exists", KEYS[1]) == 1 then return 0 end
  redis.call("set", KEYS[1], ARGV[1], "EX", ARGV[2])
  return 1
`;

// Script Lua: chỉ xóa nếu value khớp (tránh xóa nhầm lock của người khác)
const RELEASE_LUA = `
  if redis.call("get", KEYS[1]) == ARGV[1] then
    return redis.call("del", KEYS[1])
  end
  return 0
`;

export class RedisLock {
  /**
   * Danh sách Redis cần ghi lock.
   * Luôn có primary; thêm backup nếu backup đang ready.
   */
  private static targets(): Redis[] {
    const list: Redis[] = [redisPrimary];
    if (redisBackup && redisBackup.status === "ready") {
      list.push(redisBackup);
    }
    return list;
  }

  /**
   * Giữ lock ghế trên CẢ primary và backup (write-through).
   * - Primary fail → trả null (không giữ được lock).
   * - Backup conflict (đang có lock khác) → rollback primary, trả null.
   * - Backup lỗi mạng → chấp nhận (backup đang chết, không failover được).
   * TTL mặc định 30s, đủ cho transaction đặt vé.
   */
  static async acquire(
    key: string,
    ttlSeconds: number = 30,
  ): Promise<string | null> {
    const lockToken = crypto.randomUUID();
    const clients = this.targets();

    const results = await Promise.allSettled(
      clients.map((c) => c.eval(ACQUIRE_LUA, 1, key, lockToken, ttlSeconds)),
    );

    // Primary phải acquire thành công
    const primaryOk =
      results[0].status === "fulfilled" && Number(results[0].value) === 1;

    if (!primaryOk) {
      // Primary fail nhưng backup lỡ acquire được → xóa backup để tránh lock rác
      if (
        clients.length === 2 &&
        results[1].status === "fulfilled" &&
        Number(results[1].value) === 1
      ) {
        await clients[1].eval(RELEASE_LUA, 1, key, lockToken).catch(() => {});
      }
      return null;
    }

    // Primary OK — kiểm tra backup nếu có
    if (clients.length === 2) {
      const backupResult = results[1];

      // Backup trả 0 = đang có lock của người khác → rollback primary
      if (
        backupResult.status === "fulfilled" &&
        Number(backupResult.value) === 0
      ) {
        await clients[0].eval(RELEASE_LUA, 1, key, lockToken).catch(() => {});
        console.warn(
          `[REDIS LOCK] Backup đang giữ "${key}" → rollback primary, trả null`,
        );
        return null;
      }

      // Backup rejected (lỗi mạng) → chấp nhận, backup đang chết không failover được
      if (backupResult.status === "rejected") {
        console.warn(
          `[REDIS LOCK] Backup lỗi khi acquire "${key}", chấp nhận lock chỉ trên primary`,
        );
      }
    }

    return lockToken;
  }

  /**
   * Mở lock ở cả primary và backup (best-effort, xóa song song).
   * Chỉ xóa nếu value khớp token → không xóa nhầm lock của người khác.
   * Trả true nếu ít nhất primary xóa thành công.
   */
  static async release(key: string, lockToken: string): Promise<boolean> {
    const clients = this.targets();

    const results = await Promise.allSettled(
      clients.map((c) => c.eval(RELEASE_LUA, 1, key, lockToken)),
    );

    return results[0].status === "fulfilled" && Number(results[0].value) === 1;
  }
}

import type { Request, Response } from "express";
import {
  generateRegistrationOptions,
  verifyRegistrationResponse,
  generateAuthenticationOptions,
  verifyAuthenticationResponse,
} from "@simplewebauthn/server";
import prisma from "../database/prismaClient";
import { AuthService } from "../Service/authService";

const rpName = "NexusTicket";

const rpID = process.env.WEBAUTHN_RP_ID || "localhost";
const origin = process.env.WEBAUTHN_ORIGIN || "http://localhost:5173";

export class WebAuthnController {
  /**
   * 1. Tạo Challenge để đăng ký Passkey (User phải đang đăng nhập)
   */
  static async generateRegistration(req: Request, res: Response) {
    try {
      const userId = (req as any).user?.id;
      const user = await prisma.users.findUnique({
        where: { id: userId },
        include: { credentials: true },
      });

      if (!user)
        return res.status(404).json({ message: "Người dùng không tồn tại" });

      const options = await generateRegistrationOptions({
        rpName,
        rpID,
        userID: new TextEncoder().encode(userId.toString()),
        userName: user.email,
        userDisplayName: user!.fullName || user!.email,
        excludeCredentials: user.credentials.map((cred) => ({
          id: cred.credentialID,
          type: "public-key",
          transports: cred.transports
            ? (cred.transports.split(",") as any)
            : undefined,
        })),
        authenticatorSelection: {
          residentKey: "preferred",
          userVerification: "preferred",
        },
      });

      // Lưu challenge vào DB để verify sau
      await prisma.users.update({
        where: { id: userId },
        data: { currentChallenge: options.challenge },
      });

      res.json(options);
    } catch (error: any) {
      res.status(500).json({ message: error.message });
    }
  }

  /**
   * 2. Xác thực Đăng ký Passkey
   */
  static async verifyRegistration(req: Request, res: Response) {
    try {
      const userId = (req as any).user?.id;
      const { body } = req;
      const user = await prisma.users.findUnique({ where: { id: userId } });

      if (!user || !user.currentChallenge) {
        return res
          .status(400)
          .json({ message: "Không tìm thấy challenge hợp lệ" });
      }

      const verification = await verifyRegistrationResponse({
        response: body,
        expectedChallenge: user.currentChallenge,
        expectedOrigin: origin,
        expectedRPID: rpID,
      });

      if (verification.verified && verification.registrationInfo) {
        const { credential } = verification.registrationInfo;
        const { id, publicKey, counter } = credential;

        await prisma.webAuthnCredential.create({
          data: {
            userId: user.id,
            credentialID: id,
            publicKey: Buffer.from(publicKey).toString("base64url"),
            counter,
            transports: body.response.transports?.join(",") || "",
          },
        });

        return res.json({
          success: true,
          message: "Đăng ký Vân tay/Khuôn mặt thành công!",
        });
      }
      return res
        .status(400)
        .json({ success: false, message: "Xác thực thất bại!" });
    } catch (error: any) {
      return res.status(400).json({ success: false, message: error.message });
    }
  }

  /**
   * 3. Tạo Challenge để đăng nhập bằng Passkey
   */
  static async generateAuthentication(req: Request, res: Response) {
    try {
      const { email } = req.body;
      const user = await prisma.users.findUnique({
        where: { email },
        include: { credentials: true },
      });

      if (!user || user.credentials.length === 0) {
        return res
          .status(404)
          .json({ message: "Tài khoản này chưa cài đặt Vân tay/Khuôn mặt!" });
      }

      const options = await generateAuthenticationOptions({
        rpID,
        allowCredentials: user.credentials.map((cred) => ({
          id: cred.credentialID,
          type: "public-key",
          transports: cred.transports
            ? (cred.transports.split(",") as any)
            : undefined,
        })),
        userVerification: "preferred",
      });

      await prisma.users.update({
        where: { id: user.id },
        data: { currentChallenge: options.challenge },
      });

      res.json(options);
    } catch (error: any) {
      res.status(500).json({ message: error.message });
    }
  }

  /**
   * 4. Xác thực Đăng nhập Passkey & Cấp JWT
   */
  static async verifyAuthentication(req: Request, res: Response) {
    try {
      const { email, body } = req.body;
      const user = await prisma.users.findUnique({
        where: { email },
        include: { credentials: true, wallet: true },
      });

      if (!user || !user.currentChallenge) {
        return res.status(400).json({ message: "Yêu cầu không hợp lệ" });
      }

      const passkey = user.credentials.find((c) => c.credentialID === body.id);
      if (!passkey)
        return res.status(400).json({ message: "Passkey không hợp lệ!" });

      const verification = await verifyAuthenticationResponse({
        response: body,
        expectedChallenge: user.currentChallenge,
        expectedOrigin: origin,
        expectedRPID: rpID,
        credential: {
          id: passkey.credentialID,
          publicKey: Buffer.from(passkey.publicKey, "base64url"),
          counter: passkey.counter,
          transports: passkey.transports
            ? (passkey.transports.split(",") as any)
            : undefined,
        },
      });

      if (verification.verified) {
        // Cập nhật counter chống Replay Attack
        await prisma.webAuthnCredential.update({
          where: { id: passkey.id },
          data: { counter: verification.authenticationInfo.newCounter },
        });

        // Cấp phát JWT
        const tokens = AuthService.generateTokensForUser(user);
        const { password, currentChallenge, ...userWithoutPassword } = user;

        // Set Refresh Token vào HttpOnly Cookie
        res.cookie("refreshToken", tokens.refreshToken, {
          httpOnly: true,
          secure: process.env.NODE_ENV === "production",
          sameSite: "lax",
          maxAge: 7 * 24 * 60 * 60 * 1000,
          path: "/api/auth",
        });

        return res.json({
          success: true,
          accessToken: tokens.accessToken,
          user: userWithoutPassword,
        });
      }
      return res.status(400).json({ message: "Xác thực Passkey thất bại!" });
    } catch (error: any) {
      console.error("WebAuthn Login Error:", error);
      return res.status(400).json({ message: error.message });
    }
  }
}

// Hàm băm dữ liệu JSON ổn định (Sắp xếp key để chuỗi JSON luôn giống nhau)
function generateSignature(data: any): string {
  const signData = {
    requestId: data.requestId,
    userId: data.userId,
    action: data.action,
    resource: data.resource,
    resourceId: data.resourceId,
    newData: data.newData,
    oldData: data.oldData,
  };

  // Sắp xếp key theo thứ tự bảng chữ cái để đảm bảo tính nhất quán
  const sortedData = JSON.stringify(signData, Object.keys(signData).sort());
  return crypto
    .createHmac("sha256", AUDIT_SECRET)
    .update(sortedData)
    .digest("hex");
}

import prisma from "../database/prismaClient";
import { payosInstance } from "../Utils/payos";
import CircuitBreaker from "opossum";

// Cấu hình Cầu dao cho PayOS
const payosAction = async (paymentData: any) => {
  return await payosInstance.createPaymentLink(paymentData);
};

const payosBreaker = new CircuitBreaker(payosAction, {
  timeout: 10000, // PayOS thường phản hồi trong 1-2s, chờ tối đa 10s là ngắt
  errorThresholdPercentage: 50, // Ngắt nếu > 50% request lỗi
  volumeThreshold: 3, // Chỉ cần 3 request lỗi liên tiếp là ngắt
  resetTimeout: 60000, // Chờ 1 phút trước khi thử lại
});

// Khi PayOS sập, trả về lỗi ngay lập tức cho người dùng
payosBreaker.fallback(() => {
  throw new Error(
    "Cổng thanh toán PayOS đang bảo trì. Vui lòng thử lại sau 1 phút!",
  );
});

payosBreaker.on("open", () =>
  console.warn(`[Circuit Breaker] 🔴 MẠCH ĐÃ NGẮT: PayOS đang sập!`),
);

export class PaymentService {
  // 1. Tạo link thanh toán
  static async createTopUpLink(userId: number, amount: number) {
    if (amount < 10000) throw new Error("Số tiền nạp tối thiểu là 10,000 VNĐ");

    // Sinh orderCode ngẫu nhiên (9 chữ số)
    const orderCode = Number(Date.now().toString().slice(-9));
    const returnUrl = process.env.WEBAUTHN_ORIGIN || "http://localhost:5173";

    // Lưu DB trạng thái PENDING
    const transaction = await prisma.paymentTransaction.create({
      data: {
        orderCode,
        userId,
        amount,
        status: "PENDING",
      },
    });

    const paymentData = {
      orderCode,
      amount,
      description: `Nap tien vi NexusTicket`,
      cancelUrl: `${returnUrl}/customer/wallet`,
      returnUrl: `${returnUrl}/customer/wallet`,
    };

    // Gọi qua Cầu dao thay vì gọi trực tiếp payosInstance
    const paymentLinkRes = (await payosBreaker.fire(paymentData)) as any;

    // Cập nhật lại checkoutUrl vào DB
    await prisma.paymentTransaction.update({
      where: { id: transaction.id },
      data: {
        checkoutUrl: paymentLinkRes.checkoutUrl,
        paymentLinkId: paymentLinkRes.paymentLinkId,
      },
    });

    return paymentLinkRes.checkoutUrl;
  }

  // 2. Xử lý Webhook (Khi PayOS gọi về)
  static async handleWebhook(webhookBody: any) {
    const { code, success, data } = webhookBody;

    // Verify Signature (Chống hacker giả mạo)
    const isValid = payosInstance.verifyWebhookData(webhookBody);
    if (!isValid) throw new Error("Chữ ký không hợp lệ!");

    if (code === "00" && success === true) {
      const orderCode = Number(data.orderCode);
      const amountPaid = Number(data.amount);

      // Tìm giao dịch trong DB
      const transaction = await prisma.paymentTransaction.findUnique({
        where: { orderCode },
      });

      // Idempotency: Nếu đã SUCCESS rồi thì bỏ qua (PayOS có thể gọi 2 lần)
      if (!transaction || transaction.status === "SUCCESS") return;

      if (amountPaid < Number(transaction.amount)) {
        await prisma.paymentTransaction.update({
          where: { id: transaction.id },
          data: { status: "FAILED" },
        });
        return;
      }

      // Bọc trong Transaction để đảm bảo cộng tiền và update status đồng thời
      await prisma.$transaction(async (tx) => {
        // 1. Update trạng thái
        await tx.paymentTransaction.update({
          where: { id: transaction.id },
          data: { status: "SUCCESS" },
        });

        // 2. Cộng tiền vào ví
        const updatedWallet = await tx.wallets.update({
          where: { userId: transaction.userId },
          data: { balance: { increment: amountPaid } },
          select: { balance: true, id: true },
        });

        // 3. Ghi AuditLog
        await tx.auditLog.create({
          data: {
            requestId: `PAYOS_${orderCode}`,
            userId: transaction.userId,
            action: "TOP_UP",
            resource: "Wallets",
            resourceId: String(updatedWallet.id),
            oldData: { balance: Number(updatedWallet.balance) - amountPaid },
            newData: {
              amount: amountPaid,
              description: `Nạp tiền qua PayOS (Mã: ${orderCode})`,
              newBalance: Number(updatedWallet.balance),
            },
          },
        });
      });
    }
  }

import prisma from "../database/prismaClient";
import jwt from "jsonwebtoken";
import { redisPrimary } from "../Utils/redisLock";
import { MultiLevelCache } from "./cacheService";
import {
  sendNotificationEmail,
  NotificationJobData,
} from "../queues/mailQueue";

export class RevocationService {
  static async revoke(jti: string, exp: number): Promise<void> {
    if (!jti || !exp) return;
    const ttl = exp - Math.floor(Date.now() / 1000);
    if (ttl <= 0) return;
    try {
      await redisPrimary.set(`revoked:${jti}`, "1", "EX", ttl);
    } catch (err: any) {
      console.error("[REVOKE] Không lưu được jti:", err.message);
    }
  }

  static async isRevoked(jti?: string): Promise<boolean> {
    if (!jti) return false;
    try {
      return (await redisPrimary.get(`revoked:${jti}`)) === "1";
    } catch (err: any) {
      console.error("[REVOKE] Không đọc được jti:", err.message);
      return false;
    }
  }

  static async revokePair(
    accessToken?: string,
    refreshToken?: string,
  ): Promise<void> {
    for (const t of [accessToken, refreshToken]) {
      if (!t) continue;
      try {
        const decoded: any = jwt.decode(t);
        if (decoded?.jti && decoded?.exp) {
          await this.revoke(decoded.jti, decoded.exp);
        }
      } catch {
        // token hỏng thì bỏ qua
      }
    }
  }

  static async revokeBatch(batchId: string, adminUserId: number) {
    const cleanBatchId = batchId.trim();

    const emailsToSend: {
      email: string;
      type: NotificationJobData["type"];
      data: Record<string, any>;
    }[] = [];

    const affectedUserIdsList = new Set<number>();

    const result = await prisma.$transaction(async (tx) => {
      const logs = await tx.auditLog.findMany({
        where: {
          batchId: { equals: cleanBatchId, mode: "insensitive" },
          isRevoked: false,
          action: { not: "BATCH_REVOKE_EXECUTE" },
        },
      });

      if (!logs || logs.length === 0) {
        throw new Error(
          `Không tìm thấy đợt tác động "${cleanBatchId}" hoặc đã bị thu hồi!`,
        );
      }

      let revokedUsersCount = 0;

      for (const log of logs) {
        const newDataObj = (
          typeof log.newData === "string"
            ? JSON.parse(log.newData)
            : log.newData
        ) as any;

        if (log.action === "MASS_GIFT_EXECUTE") {
          const giftedAmount = Number(newDataObj?.amountPerUser || 0);
          const targetUserIds: number[] = newDataObj?.affectedUserIds || [];

          if (
            giftedAmount > 0 &&
            Array.isArray(targetUserIds) &&
            targetUserIds.length > 0
          ) {
            await tx.wallets.updateMany({
              where: { userId: { in: targetUserIds } },
              data: { balance: { decrement: giftedAmount } },
            });

            const affectedWallets = await tx.wallets.findMany({
              where: { userId: { in: targetUserIds } },
              select: {
                id: true,
                userId: true,
                balance: true,
                user: { select: { email: true } },
              },
            });

            const revokeWalletLogs = affectedWallets.map((w) => {
              const currentBalance = Number(w.balance);
              targetUserIds.forEach((id) => affectedUserIdsList.add(id));

              if (w.user?.email) {
                emailsToSend.push({
                  email: w.user.email,
                  type: "ADMIN_REVOKE_GIFT",
                  data: {
                    amount: giftedAmount,
                    reason: `Admin thu hồi quà tặng (Đợt: ${cleanBatchId})`,
                  },
                });
              }

              return {
                requestId: `REVOKE_GIFT_${Date.now()}_${w.userId}`,
                userId: w.userId,
                action: "MASS_GIFT_REVOKED",
                batchId: cleanBatchId,
                resource: "Wallets",
                resourceId: String(w.id),
                oldData: { balance: currentBalance + giftedAmount },
                newData: {
                  amount: -giftedAmount,
                  description: `Admin thu hồi quà tặng tiền ví (Đợt: ${cleanBatchId})`,
                  newBalance: currentBalance,
                },
                isRevoked: false,
              };
            });

            await tx.auditLog.createMany({
              data: revokeWalletLogs,
            });

            revokedUsersCount += targetUserIds.length;
          }
        } else if (
          log.action === "BOOK_TICKET_PAYMENT" ||
          log.action === "BOOK_TICKET_BATCH" ||
          log.action === "MASS_GIFT_TICKET" ||
          log.resource === "Tickets"
        ) {
          const tripId = Number(newDataObj?.tripId);
          const seats: number[] = Array.isArray(newDataObj?.seats)
            ? newDataObj.seats.map((s: any) => Number(s))
            : [];
          const logUserId = log.userId;

          if (tripId && seats.length > 0 && logUserId) {
            const ticketsToRevoke = await tx.tickets.findMany({
              where: {
                tripId: tripId,
                seatNumber: { in: seats },
                userId: logUserId,
                status: "CONFIRMED",
              },
            });

            if (ticketsToRevoke.length > 0) {
              const ticketIds = ticketsToRevoke.map((t) => t.id);

              await tx.tickets.updateMany({
                where: { id: { in: ticketIds } },
                data: { status: "REVOKED_BY_ADMIN" },
              });

              try {
                await MultiLevelCache.invalidate(`trips:detail:${tripId}`);
                await MultiLevelCache.invalidate("trips:all:active");
              } catch (cacheErr) {
                console.warn("Xóa cache thất bại:", cacheErr);
              }

              const actualPaidAmount = Math.abs(
                Number(newDataObj?.amount || 0),
              );

              const user = await tx.users.findUnique({
                where: { id: logUserId },
                select: { email: true },
              });

              if (user?.email) {
                affectedUserIdsList.add(logUserId);
                emailsToSend.push({
                  email: user.email,
                  type: "ADMIN_REVOKE_BATCH",
                  data: {
                    userId: logUserId,
                    batchId: cleanBatchId,
                    reason: `Hệ thống thu hồi vé theo đợt (${cleanBatchId})`,
                  },
                });
              }

              if (actualPaidAmount > 0) {
                const updatedWallet = await tx.wallets.update({
                  where: { userId: logUserId },
                  data: { balance: { increment: actualPaidAmount } },
                  select: { id: true, balance: true },
                });

                const currentBalance = Number(updatedWallet.balance);

                await tx.auditLog.create({
                  data: {
                    requestId: `REFUND_${Date.now()}_${logUserId}`,
                    userId: logUserId,
                    action: "REFUND_TICKET_PAYMENT",
                    batchId: cleanBatchId,
                    resource: "Wallets",
                    resourceId: String(updatedWallet.id),
                    oldData: { balance: currentBalance - actualPaidAmount },
                    newData: {
                      amount: actualPaidAmount,
                      description: `Hoàn tiền ${ticketsToRevoke.length} vé chuyến #${tripId} (Thu hồi đợt lỗi hệ thống)`,
                      newBalance: currentBalance,
                      tripId,
                      seats,
                    },
                    isRevoked: false,
                  },
                });
              }

              revokedUsersCount += ticketsToRevoke.length;
            }
          }
        }

        await tx.auditLog.update({
          where: { id: log.id },
          data: { isRevoked: true },
        });
      }

      await tx.auditLog.create({
        data: {
          requestId: `REVOKE_SUMMARY_${Date.now()}`,
          userId: adminUserId,
          action: "BATCH_REVOKE_EXECUTE",
          batchId: cleanBatchId,
          resource: "System",
          resourceId: cleanBatchId,
          newData: {
            revokedTotal: revokedUsersCount,
            performedByAdminId: adminUserId,
            affectedUserIds: Array.from(affectedUserIdsList),
            targetType: "ALL",
          },
        },
      });

      return { success: true, revokedCount: revokedUsersCount };
    });

    emailsToSend.forEach((payload) => {
      sendNotificationEmail(payload).catch((err) => {
        console.error(
          `[REVOKE] Lỗi đẩy job gửi mail cho ${payload.email}:`,
          err?.message,
        );
      });
    });

    return result;
  }
}

import prisma from "../database/prismaClient";

export class TicketService {
  // 1. Lấy danh sách vé cho Admin (có tìm kiếm & phân trang)
  static async getAllTicketsForAdmin(
    page: number = 1,
    limit: number = 15,
    search?: string,
  ) {
    const skip = (page - 1) * limit;

    const whereCondition: any = {};
    if (search) {
      const searchNum = Number(search);
      whereCondition.OR = [
        { id: isNaN(searchNum) ? undefined : searchNum },
        { user: { email: { contains: search, mode: "insensitive" } } },
        { trip: { route: { contains: search, mode: "insensitive" } } },
        { batchId: { contains: search, mode: "insensitive" } },
      ].filter((cond) => Object.values(cond)[0] !== undefined);
    }

    const [tickets, total] = await prisma.$transaction([
      prisma.tickets.findMany({
        where: whereCondition,
        include: {
          user: { select: { email: true, fullName: true } },
          trip: { select: { route: true, price: true } },
        },
        orderBy: { createdAt: "desc" },
        skip,
        take: limit,
      }),
      prisma.tickets.count({ where: whereCondition }),
    ]);

    // Đánh dấu các vé bất thường (giá thực trả < giá gốc hoặc = 0)
    const formattedTickets = tickets.map((t) => {
      const isAnomaly =
        t.status === "CONFIRMED" &&
        (Number(t.paidAmount || 0) < Number(t.trip.price) ||
          Number(t.paidAmount) === 0);
      return { ...t, isAnomaly };
    });

    return {
      data: formattedTickets,
      meta: { total, page, limit, totalPages: Math.ceil(total / limit) },
    };
  }

  // 2. Dùng AI quét bất thường (Tìm vé 0đ hoặc sai giá)
  static async scanAnomalies() {
    const anomalies = await prisma.$queryRaw`
      SELECT t.id, t."userId", t."tripId", t."paidAmount", tr.price AS "expectedPrice", tr.route
      FROM "Tickets" t
      JOIN "Trips" tr ON t."tripId" = tr.id
      WHERE t.status = 'CONFIRMED' 
        AND (t."paidAmount" IS NULL OR t."paidAmount" < tr.price OR t."paidAmount" = 0)
      ORDER BY t."createdAt" DESC
      LIMIT 20;
    `;

    if (!anomalies || (anomalies as any[]).length === 0) {
      return {
        anomalies: [],
        aiAnalysis: "Hệ thống không phát hiện bất thường nào.",
      };
    }

    try {
      const prompt = `Bạn là chuyên gia tài chính. Hãy phân tích danh sách các vé bị lỗi giá sau đây và dự đoán nguyên nhân (VD: Lỗi nhập giá 0đ, bug code, hay vé tặng?): ${JSON.stringify(anomalies)}. Trả lời ngắn gọn trong 1 đoạn văn.`;

      // SỬA LỖI: Dùng đúng tên hàm generateWithFallback
      const aiResponse = await AIProxyService.generateWithFallback(prompt);

      return { anomalies, aiAnalysis: aiResponse };
    } catch (error: any) {
      return {
        anomalies,
        aiAnalysis:
          "Không thể gọi AI phân tích lúc này, nhưng danh sách vé lỗi đã được tìm thấy.",
      };
    }
  }
}

import prisma from "../database/prismaClient";

export class WalletService {
  static async getBalance(userId: number) {
    const wallet = await prisma.wallets.findUnique({
      where: { userId },
    });

    if (!wallet) {
      const error: any = new Error("Không tìm thấy ví của người dùng này!");
      error.statusCode = 404;
      throw error;
    }

    return wallet;
  }

  static async topUp(
    userId: number,
    amount: number,
    options?: {
      requestId?: string | undefined;
      batchId?: string | undefined;
      action?: string | undefined;
      ipAddress?: string | undefined;
    },
  ) {
    if (amount <= 0) {
      const error: any = new Error("Số tiền nạp phải lớn hơn 0!");
      error.statusCode = 400;
      throw error;
    }

    return await prisma.$transaction(async (tx) => {
      const wallet = await tx.wallets.findUnique({
        where: { userId },
      });

      if (!wallet) {
        const error: any = new Error("Không tìm thấy ví của người dùng này!");
        error.statusCode = 404;
        throw error;
      }

      if (wallet.isLocked) {
        const error: any = new Error(
          "Ví này đang bị khóa do vi phạm hoặc chờ đối soát. Không thể nạp tiền!",
        );
        error.statusCode = 403;
        throw error;
      }

      const updatedWallet = await tx.wallets.update({
        where: { userId },
        data: {
          balance: {
            increment: amount,
          },
        },
      });

      await tx.auditLog.create({
        data: {
          requestId: options?.requestId || `TOPUP_${Date.now()}`,
          userId: userId,
          action: options?.action || "TOP_UP",
          batchId: options?.batchId || null,
          resource: "Wallets",
          resourceId: String(updatedWallet.id),
          oldData: { balance: wallet.balance },
          newData: {
            amount: amount,
            giftAmount: amount,
            description: `Nạp tiền vào ví: +${amount.toLocaleString("vi-VN")} VNĐ`,
            newBalance: updatedWallet.balance,
          },
          ipAddress: options?.ipAddress || null,
        },
      });

      return updatedWallet;
    });
  }

  static async getWalletTransactions(
    userId: number,
    params: { page?: number; limit?: number } = {},
  ) {
    const page = params.page || 1;
    const limit = params.limit || 10;
    const skip = (page - 1) * limit;

    const wallet = await prisma.wallets.findUnique({
      where: { userId },
    });

    const walletIdStr = wallet ? String(wallet.id) : null;
    const currentBalance = wallet ? Number(wallet.balance) : 0;

    if (!walletIdStr) {
      return {
        currentBalance: 0,
        transactions: [],
        pagination: {
          total: 0,
          page,
          limit,
          totalPages: 0,
        },
      };
    }

    const whereCondition = {
      OR: [
        {
          userId: userId,
          action: {
            in: [
              "TOP_UP",
              "BOOK_TICKET_PAYMENT",
              "CANCEL_TICKET_REFUND",
              "SYSTEM_GIFT_BALANCE",
              "MASS_GIFT_RECEIVED",
              "MASS_GIFT_REVOKED",
              "REFUND_TICKET_PAYMENT",
              "SYSTEM_AUTO_DEBIT_AND_FREEZE",
              "SYSTEM_ALERT_BALANCE_MISMATCH",
            ],
          },
        },
        {
          resource: "Wallets",
          resourceId: walletIdStr,
          action: {
            in: [
              "MASS_GIFT_WALLET",
              "MASS_GIFT_RECEIVED",
              "MASS_GIFT_REVOKED",
              "REVOKE_MASS_GIFT",
              "REVOKE_BATCH",
              "ADMIN_ADJUST_BALANCE",
              "REFUND_TICKET_PAYMENT",
              "SYSTEM_AUTO_DEBIT_AND_FREEZE",
            ],
          },
        },
      ],
    };

    const [transactions, total] = await prisma.$transaction([
      prisma.auditLog.findMany({
        where: whereCondition,
        skip,
        take: limit,
        orderBy: { createdAt: "desc" },
      }),
      prisma.auditLog.count({
        where: whereCondition,
      }),
    ]);

    return {
      currentBalance,
      transactions,
      pagination: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    };
  }
}

import { redisPrimary } from "./redisLock";

export class RateLimiter {
  /**
   * Thuật toán Fixed Window: Đếm số request trong 1 khoảng thời gian
   * @param key Khóa định danh (VD: IP hoặc UserID)
   * @param limit Số request tối đa
   * @param windowInSeconds Thời gian (giây)
   */
  static async consume(
    key: string,
    limit: number,
    windowInSeconds: number,
  ): Promise<boolean> {
    const redisKey = `ratelimit:${key}`;

    // Script Lua đảm bảo tính nguyên tử (Atomic)
    const luaScript = `
      local current = redis.call("INCR", KEYS[1])
      if current == 1 then
        redis.call("EXPIRE", KEYS[1], ARGV[1])
      end
      if current > tonumber(ARGV[2]) then
        return 0
      end
      return 1
    `;

    try {
      const result = await redisPrimary.eval(
        luaScript,
        1,
        redisKey,
        windowInSeconds,
        limit,
      );

      return result === 1;
    } catch (error) {
      console.error("[RATE LIMITER] Lỗi Redis:", error);
      return true; // Fail-open: Nếu Redis lỗi, cho qua để không sập web
    }
  }
}

export const initTicketAuditorCron = () => {
  cron.schedule("*/15 * * * *", async () => {

    isRunning = true;
    console.log("[AUDITOR CRON] Bắt đầu quét vé bất thường tự động...");

    try {
      // 1. Tìm các vé lỗi (giới hạn 10 vé để tránh quá tải)
      const anomalies = await prisma.$queryRaw`
        SELECT t.id, t."userId", t."tripId", t."paidAmount", tr.price AS "expectedPrice", tr.route
        FROM "Tickets" t
        JOIN "Trips" tr ON t."tripId" = tr.id
        WHERE t.status = 'CONFIRMED' 
          AND (t."paidAmount" IS NULL OR t."paidAmount" < tr.price OR t."paidAmount" = 0)
        ORDER BY t."createdAt" DESC
        LIMIT 10;
      `;

      if (!anomalies || (anomalies as any[]).length === 0) {
        isRunning = false;
        return;
      }

      console.log(
        `[AUDITOR CRON] Phát hiện ${(anomalies as any[]).length} vé lỗi. Đang nhờ AI phân tích...`,

        "Hệ thống AI đang quá tải, không thể phân tích lúc này. Vui lòng kiểm tra thủ công.";

      try {
        const prompt = `Bạn là chuyên gia tài chính. Hãy phân tích danh sách các vé bị lỗi giá sau đây và dự đoán nguyên nhân (VD: Lỗi nhập giá 0đ, bug code, hay vé tặng?): ${JSON.stringify(anomalies)}. Trả lời ngắn gọn trong 1 đoạn văn.`;

        );

        aiAnalysis = await Promise.race([aiPromise, timeoutPromise]);
      } catch (aiError: any) {
        console.error("[AUDITOR CRON] Lỗi phân tích AI:", aiError.message);
        // Vẫn giữ lại giá trị mặc định của biến aiAnalysis nếu AI sập
      }

      // 3. Lưu thông báo vào AuditLog (LUÔN LƯU, kể cả khi AI lỗi)

        data: {
          requestId: `AUDITOR_AI_${Date.now()}`,

          resource: "System",
          resourceId: "TICKET_AUDIT",
          newData: {
            message: `Phát hiện ${(anomalies as any[]).length} vé bất thường!`,
            aiAnalysis: aiAnalysis,
            anomalies: anomalies,
          },
        },
      });

      console.log("[AUDITOR CRON] Đã lưu báo cáo cảnh báo lên hệ thống.");
    } catch (error: any) {
      console.error("[AUDITOR CRON] Lỗi quét DB:", error.message);
    } finally {
      isRunning = false;
    }
  });

import crypto from "crypto";

const AUDIT_SECRET =
  process.env.AUDIT_SECRET_KEY || "NexusTicket_Audit_Secret_2026";

export class AuditSigner {
  static generateSignature(data: any): string {
    // Sắp xếp các key một cách đệ quy để đảm bảo tính nhất quán
    const sortObject = (obj: any): any => {
      if (obj === null || typeof obj !== "object") return obj;
      if (Array.isArray(obj)) return obj.map(sortObject);
      return Object.keys(obj)
        .sort()
        .reduce((acc, key) => {
          acc[key] = sortObject(obj[key]);
          return acc;
        }, {} as any);
    };

    const signData = {
      requestId: data.requestId,
      userId: data.userId,
      action: data.action,
      resource: data.resource,
      resourceId: data.resourceId,
      newData: data.newData,
      oldData: data.oldData,
    };

    const sortedData = JSON.stringify(sortObject(signData));
    return crypto
      .createHmac("sha256", AUDIT_SECRET)
      .update(sortedData)
      .digest("hex");
  }

  static verifySignature(data: any, signature: string): boolean {
    if (!signature) return false;
    const expectedSignature = this.generateSignature(data);
    return expectedSignature === signature;
  }
}

import type { Request, Response, NextFunction } from "express";
import { RateLimiter } from "../Utils/rateLimiter";

export const rateLimit = (limit: number, windowInSeconds: number) => {
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      const ip =
        (req.headers["x-forwarded-for"] as string)?.split(",")[0]?.trim() ||
        req.socket.remoteAddress ||
        "127.0.0.1";

      const isAllowed = await RateLimiter.consume(ip, limit, windowInSeconds);

      if (!isAllowed) {
        return res.status(429).json({
          success: false,
          message: "Bạn đã thao tác quá nhanh. Vui lòng thử lại sau!",
        });
      }

      next();
    } catch (error) {
      next();
    }
  };
};

# NexusTicket – Hệ Thống Đặt Vé Phân Tán (High Availability, AIOps & Passwordless Auth)

NexusTicket là hệ thống đặt vé xe trực tuyến toàn diện, được thiết kế theo mô hình **Modular Monolith**. Dự án hướng tới mục tiêu trở thành một **Sản phẩm Nghiên cứu (SPNC)** với kiến trúc bảo mật đa lớp, khả năng chịu tải cao, và tích hợp Trí tuệ Nhân tạo (AI) sâu vào khâu vận hành (AIOps).

- **Backend:** Node.js, Express, TypeScript, Prisma ORM, PostgreSQL.
- **Frontend:** React, Vite, TailwindCSS, React Query.
- **Hạ tầng & Tiện ích:** Vercel (API & Daemons), Docker, Dual Redis Cluster (HA), Multi-Level Cache, BullMQ, PayOS Payment Gateway, Cloudinary.
- **Bảo mật & Chịu lỗi:** Passkeys (WebAuthn), JWT Rotation & Revocation, HMAC Audit Logs, Rate Limiting, Helmet, Circuit Breaker (Opossum).
- **Kiểm thử:** Jest, Playwright (E2E), k6 (Load Test).

---

## ⚠️ CẢNH BÁO QUAN TRỌNG VỀ THANH TOÁN

Hệ thống hiện tại đã được cấu hình cổng thanh toán **PayOS ở môi trường Production (Thanh toán thật)**. Mọi giao dịch quét mã QR/VietQR sẽ **trừ tiền thật** trong tài khoản ngân hàng của bạn.

- **Mục đích:** Hệ thống dùng cho mục đích nghiên cứu và test luồng thanh toán thực tế.
- **Biện pháp bảo vệ:** Giao diện Frontend đã được tích hợp **Banner cảnh báo** và số điện thoại liên hệ. Nếu lỡ phát sinh giao dịch chuyển tiền nhầm, vui lòng liên hệ ngay số **0363.489.863** để được hỗ trợ hoàn tiền.

## 🏗️ 1. Kiến Trúc Hệ Thống (System Architecture)

Hệ thống được tổ chức theo luồng: **Client → API Gateway → Domain Services → Operations & State**.

```text

             │
             ▼

             │ (REST API)

   ├── AuthService (Passkeys + JWT Rotation + Revocation)
   ├── OtpService (BullMQ / Resend Worker)
   ├── TripService (Hybrid Search: Text + Gemini Vector)

   ├── PaymentService (PayOS Integration & Webhook HMAC Verify)
```

## 🔐 2. Tính Năng Bảo Mật & Kiểm Soát Tài Chính (Zero-Trust)

### 2.1. Xác Thực Không Mật Khẩu (Passkeys / WebAuthn)

- Hỗ trợ đăng nhập bằng Vân tay, Face ID hoặc Windows Hello. Loại bỏ hoàn toàn rủi ro Phishing vì khóa riêng tư không rời khỏi thiết bị.

### 2.2. JWT Hiện Đại với Refresh Token Rotation & Revocation

- **Access Token** (15 phút) lưu trong RAM, **Refresh Token** (7 ngày) lưu trong **HttpOnly Cookie** (chống XSS).
- **Revocation List (Redis):** Khi logout, token lập tức bị vô hiệu hóa trên Redis.

### 2.3. Bảo Vệ Dòng Tiền & Khóa Ví Tự Động (AIOps)

- **Khóa ví tự động:** Module Đối soát (Cron) tự động quét DB, nếu phát hiện số dư lệch so với Lịch sử (AuditLog) sẽ lập tức gán cờ `isLocked = true` và ghi log cảnh báo.
- **Chặn giao dịch:** Mọi API Mua vé và Nạp tiền đều bắt buộc kiểm tra `isLocked`. Nếu ví bị khóa, trả về lỗi `403 Forbidden`.

### 2.4. Toàn Vẹn Dữ Liệu (HMAC Audit Logs)

- Bảng `AuditLog` được bảo vệ bằng **chữ ký số HMAC-SHA256**. Nếu có bất kỳ ai (kể cả Admin) can thiệp sửa đổi số tiền trong DB, chữ ký sẽ không còn khớp, hệ thống AIOps sẽ phát hiện và đưa ra cảnh báo đỏ.

### 2.5. Bảo Mật Header với Helmet

- Tích hợp `helmet` để giấu header `X-Powered-By`, chặn các lỗ hổng XSS cơ bản và bảo vệ thông tin máy chủ khỏi các cuộc tấn công dò quét.

## ⚡ 3. Các Tính Năng Kỹ Thuật Cốt Lõi

- **High Availability:** Dual Redis Cluster tự động chuyển vùng (Failover) khi Primary gặp sự cố.
- **Multi-Level Cache:** Kết hợp LRU Cache (L1 - RAM) và Redis (L2) giúp truy vấn danh mục chuyến xe với độ trễ cực thấp.
- **Chống Race Condition:** Sử dụng **Redis Distributed Lock** kết hợp **Postgres Advisory Lock** trong `prisma.$transaction` đảm bảo không xảy ra tình trạng overselling (bán trùng ghế).
- **Tìm kiếm AI (Hybrid Search):** Kết hợp tìm kiếm từ khóa (ILIKE) và tìm kiếm ngữ nghĩa (Gemini Vector) với cơ chế Fallback thông minh (tự động bỏ lọc thời gian nếu không có kết quả).

### 3.5. Cầu Dao Điện (Circuit Breaker) & Chịu Lỗi

- **Circuit Breaker (Opossum):** Tích hợp cho các dịch vụ bên thứ 3 (Gemini AI, PayOS). Nếu dịch vụ sập liên tục (lỗi 503), mạch sẽ ngắt (Open) và trả về Fallback ngay lập tức (0ms) thay vì bắt khách hàng chờ 20s. Sau 1 phút, mạch tự động đóng lại (Half-Open) để thử kết nối.
- **Graceful Degradation:** Nếu AI sập, tính năng tìm kiếm tự động chuyển về tìm kiếm Text (ILIKE) cơ bản. Nếu Webhook PayOS đến muộn, hệ thống vẫn xử lý Idempotency (chống cộng tiền 2 lần).

### 3.6. Chế Độ Bảo Trì Khẩn Cấp (Emergency Maintenance)

- Admin có thể bật/tắt chế độ bảo trì 1-Click từ trang **System Health**.
- Khi bật, hệ thống lập tức chặn mọi API Mua vé/Nạp tiền (trả về lỗi 503) để đảm bảo an toàn dữ liệu trong lúc nâng cấp Database.
- **Đặc biệt:** Webhook PayOS vẫn được xuyên qua để cộng tiền nếu khách đã lỡ chuyển khoản.

import type { Request, Response, NextFunction } from "express";
import { BookingService } from "../Service/bookingService";
import prisma from "../database/prismaClient";

const catchAsync = (
  fn: (req: Request, res: Response, next: NextFunction) => Promise<any>,
) => {
  return (req: Request, res: Response, next: NextFunction) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
};

export class BookingController {
  static createBooking = catchAsync(async (req: Request, res: Response) => {
    const userId = (req as any).user?.id || (req as any).user?.userId;
    if (!userId) {
      return res
        .status(401)
        .json({ success: false, message: "Chưa đăng nhập!" });
    }

    const { tripId, seatNumber, seatNumbers, paymentMethod, batchId } =
      req.body;

    const seatsToBook: number[] = (
      seatNumbers || (seatNumber ? [seatNumber] : [])
    ).map(Number);

    if (seatsToBook.length === 0) {
      return res
        .status(400)
        .json({ success: false, message: "Vui lòng chọn ghế!" });
    }

    const results = await BookingService.bookMultipleSeats(
      Number(userId),
      Number(tripId),
      seatsToBook,
      paymentMethod || "WALLET",
      batchId,
    );

    return res.status(201).json({
      success: true,
      message: `Đặt thành công ${seatsToBook.length} vé!`,
      data: results.length === 1 ? results[0] : results,
    });
  });

  static cancelBooking = catchAsync(async (req: Request, res: Response) => {
    const userId = (req as any).user?.id || (req as any).user?.userId;
    const ticketId = Number(req.params.id);

    if (!userId) {
      return res
        .status(401)
        .json({ success: false, message: "Chưa đăng nhập!" });
    }

    const canceledTicket = await BookingService.cancelTicket(userId, ticketId);

    return res.json({
      success: true,
      message: "Hủy vé thành công và đã hoàn tiền vào ví (nếu có)!",
      data: canceledTicket,
    });
  });

  static getUserTickets = catchAsync(async (req: Request, res: Response) => {
    const userIdFromToken = (req as any).user?.id || (req as any).user?.userId;
    const role = (req as any).user?.role;

    let targetUserId = Number(userIdFromToken);
    if (role === "ADMIN" && req.params.userId) {
      targetUserId = Number(req.params.userId);
    }

    if (!targetUserId) {
      return res
        .status(400)
        .json({ success: false, message: "Mã người dùng không hợp lệ!" });
    }

    const page = req.query.page ? Number(req.query.page) : 1;
    const limit = req.query.limit ? Number(req.query.limit) : 5;
    const skip = (page - 1) * limit;

    const [tickets, total] = await prisma.$transaction([
      prisma.tickets.findMany({
        where: { userId: targetUserId },
        include: { trip: true },
        orderBy: { createdAt: "desc" },
        skip,
        take: limit,
      }),
      prisma.tickets.count({ where: { userId: targetUserId } }),
    ]);

    return res.status(200).json({
      success: true,
      data: tickets,
      meta: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    });
  });

  static adminRevokeBooking = catchAsync(
    async (req: Request, res: Response) => {
      const ticketId = Number(req.params.id);
      const adminUserId = (req as any).user?.id || (req as any).user?.userId;
      const { reason } = req.body;

      const revokedTicket = await BookingService.adminRevokeTicket(
        ticketId,
        Number(adminUserId),
        reason || "Admin thu hồi vé",
      );

      return res.json({
        success: true,
        message: "Đã thu hồi vé thành công!",
        data: revokedTicket,
      });
    },
  );
}

import { Queue } from "bullmq";
import Redis from "ioredis";

export interface MailJobData {
  email: string;
  code: string;
  type: "REGISTER" | "FORGOT_PASSWORD";
}

export interface NotificationJobData {
  email: string;
  type:
    | "BOOKING_SUCCESS"
    | "BOOKING_SUCCESS_BATCH"
    | "BOOKING_CANCEL"
    | "ADMIN_REVOKE"
    | "ADMIN_REVOKE_BATCH"
    | "ADMIN_REVOKE_MULTIPLE"
    | "WALLET_LOCKED_AI"
    | "ADMIN_GIFT"
    | "ADMIN_REVOKE_GIFT";
  data: Record<string, any>;
}

export type AllJobData = MailJobData | NotificationJobData;

const redisUrl = process.env.REDIS_URL || "redis://localhost:6379";
const isTls = redisUrl.startsWith("rediss://");

const connection = new Redis(redisUrl, {
  maxRetriesPerRequest: null,
  enableReadyCheck: false,
  keepAlive: 10000,
  tls: isTls ? { rejectUnauthorized: false } : undefined,
});

export const mailQueue = new Queue<AllJobData>("mail-queue", { connection });

export async function sendNotificationEmail(payload: {
  email: string;
  type: NotificationJobData["type"];
  data: Record<string, any>;
}) {
  await mailQueue.add("send-notification-job", payload, {
    attempts: 3,
    backoff: { type: "exponential", delay: 3000 },
    removeOnComplete: 200,
    removeOnFail: 500,
  });
}

export async function scheduleNotificationWithReset(
  payload: {
    email: string;
    type: NotificationJobData["type"];
    data: Record<string, any>;
  },
  opts: { delayMs: number; jobId: string },
) {
  const safeJobId = opts.jobId.replace(/:/g, "-");

  try {
    const existing = await mailQueue.getJob(safeJobId);
    if (existing) await existing.remove();
  } catch (err: any) {
    console.warn(`[MAIL] Không thể xóa job cũ ${safeJobId}:`, err?.message);
  }

  try {
    await mailQueue.add("send-notification-job", payload, {
      delay: opts.delayMs,

  static async invalidateTripCaches(tripId?: number): Promise<void> {

    const availableSeats = Math.max(0, totalSeats - bookedSeatNumbers.length);

      availableSeats,

       RETURNING id`,

      trimmedDesc || null,

      finalSeats,
      vectorString,

  static async searchTripsSemanticLogic(prompt: string = "", limit = 10) {
    const safePrompt = typeof prompt === "string" ? prompt : "";
    const cleanPrompt = safePrompt.trim().toLowerCase();

    if (
      !cleanPrompt ||
      cleanPrompt.includes("tất cả") ||
      cleanPrompt.includes("danh sách")
    ) {
      return await this.getAllTripsLogic();
    }

    const now = new Date();

    // 1. Xử lý từ khóa thời gian (Mở rộng thêm "đêm", "tối nay", v.v.)
    let timeConditionPrisma: any = { departureAt: { gt: now } };
    let timeConditionSQL = "";
    let hasTimeFilter = false;

    if (
      cleanPrompt.includes("tối") ||
      cleanPrompt.includes("đêm") ||
      cleanPrompt.includes("khuya")
    ) {
      const startEvening = new Date(now);
      startEvening.setHours(18, 0, 0, 0);
      const endEvening = new Date(now);
      endEvening.setHours(23, 59, 59, 999);
      const searchStart = now > startEvening ? now : startEvening;

      timeConditionPrisma = {
        departureAt: { gte: searchStart, lte: endEvening },
      };
      timeConditionSQL = `AND EXTRACT(HOUR FROM "departureAt") >= 18`;
      hasTimeFilter = true;
    } else if (cleanPrompt.includes("sáng")) {
      const startMorning = new Date(now);
      startMorning.setHours(0, 0, 0, 0);
      const endMorning = new Date(now);
      endMorning.setHours(11, 59, 59, 999);
      const searchStart =
        now > endMorning
          ? new Date(now.setDate(now.getDate() + 1))
          : now > startMorning
            ? now
            : startMorning;
      searchStart.setHours(0, 0, 0, 0);

      timeConditionPrisma = { departureAt: { gt: searchStart } };
      timeConditionSQL = `AND EXTRACT(HOUR FROM "departureAt") < 12`;
      hasTimeFilter = true;
    } else if (cleanPrompt.includes("chiều")) {
      const startAfternoon = new Date(now);
      startAfternoon.setHours(12, 0, 0, 0);
      const endAfternoon = new Date(now);
      endAfternoon.setHours(17, 59, 59, 999);
      const searchStart = now > startAfternoon ? now : startAfternoon;

      timeConditionPrisma = {
        departureAt: { gte: searchStart, lte: endAfternoon },
      };
      timeConditionSQL = `AND EXTRACT(HOUR FROM "departureAt") >= 12 AND EXTRACT(HOUR FROM "departureAt") < 18`;
      hasTimeFilter = true;
    }

    // 2. Lọc bỏ các từ nhiễu (CHỈ dùng cho SQL ILIKE, giữ nguyên câu cho AI)
    const stopWords = [
      "tôi",
      "mình",
      "muốn",
      "cần",
      "tìm",
      "xe",
      "đi",
      "vé",
      "chuyến",
      "có",
      "không",
      "và",
      "thì",
      "là",
      "ở",
      "từ",
      "đến",
      "về",
      "những",
      "các",
      "một",
      "cho",
    ];
    const searchTerms = cleanPrompt
      .split(/\s+/)
      .filter((w) => !stopWords.includes(w));
    const keywordSearch = searchTerms.length > 0 ? searchTerms.join(" ") : "";

    const resultMap = new Map<number, any>();

    // 3. TÌM KIẾM TEXT (Lần 1: Có bộ lọc thời gian)
    if (searchTerms.length > 0) {
      const whereCondition: any = { ...timeConditionPrisma };
      whereCondition.AND = searchTerms.map((term) => ({
        OR: [
          { route: { contains: term, mode: "insensitive" } },
          { description: { contains: term, mode: "insensitive" } },
        ],
      }));

      const keywordTrips = await prisma.trips.findMany({
        where: whereCondition,
        take: limit,
        orderBy: { departureAt: "asc" },
        include: {
          tickets: {
            where: { status: { in: ["HELD", "PENDING", "CONFIRMED"] } },
            select: { seatNumber: true },
          },
        },
      });

      keywordTrips.forEach((trip) =>
        resultMap.set(trip.id, this.formatTripWithSeats(trip)),
      );
    }

    // 4. NẾU ÁP DỤNG THỜI GIAN MÀ KHÔNG CÓ KẾT QUẢ -> TỰ ĐỘNG BỎ LỌC THỜI GIAN (Fallback)
    if (hasTimeFilter && resultMap.size === 0 && searchTerms.length > 0) {
      console.log(
        "[SEARCH] Không có chuyến đúng giờ, tự động mở rộng tìm kiếm bỏ qua thời gian.",
      );
      const whereCondition: any = { departureAt: { gt: now } };
      whereCondition.AND = searchTerms.map((term) => ({
        OR: [
          { route: { contains: term, mode: "insensitive" } },
          { description: { contains: term, mode: "insensitive" } },
        ],
      }));

      const fallbackTrips = await prisma.trips.findMany({
        where: whereCondition,
        take: limit,
        orderBy: { departureAt: "asc" },
        include: {
          tickets: {
            where: { status: { in: ["HELD", "PENDING", "CONFIRMED"] } },
            select: { seatNumber: true },
          },
        },
      });
      fallbackTrips.forEach((trip) =>
        resultMap.set(trip.id, this.formatTripWithSeats(trip)),
      );
    }

    // 5. SEMANTIC SEARCH (AI) - Chạy nếu chưa đủ số lượng
    if (keywordSearch && resultMap.size < limit) {
      try {
        // QUAN TRỌNG: Đưa NGUYÊN CÂU GỐC (cleanPrompt) vào AI để nó hiểu ngữ cảnh
        // Ví dụ: "tìm xe đi đà lạt buổi tối" -> AI sẽ tự map "buổi tối" với các chuyến 18h+
        const queryVector = await this.getEmbedding(cleanPrompt);
        const vectorString = `[${queryVector.join(",")}]`;

        const SIMILARITY_THRESHOLD = 0.7;

        let semanticTrips: any[] = await prisma.$queryRawUnsafe(
          `
          SELECT id, route, "departureAt", price, "totalSeats", description,
                 1 - (embedding <=> $1::vector) AS similarity
          FROM "Trips"
          WHERE "departureAt" > NOW()
            ${timeConditionSQL}
            AND 1 - (embedding <=> $1::vector) > $2
          ORDER BY similarity DESC
          LIMIT $3
        `,
          vectorString,
          SIMILARITY_THRESHOLD,
          limit,
        );

        // Nếu AI có thời gian mà vẫn rỗng, thử AI không có thời gian
        if (semanticTrips.length === 0 && hasTimeFilter) {
          semanticTrips = await prisma.$queryRawUnsafe(
            `
            SELECT id, route, "departureAt", price, "totalSeats", description,
                   1 - (embedding <=> $1::vector) AS similarity
            FROM "Trips"
            WHERE "departureAt" > NOW()
              AND 1 - (embedding <=> $1::vector) > $2
            ORDER BY similarity DESC
            LIMIT $3
          `,
            vectorString,
            SIMILARITY_THRESHOLD,
            limit,
          );
        }

        // Gộp kết quả AI vào Map
        if (semanticTrips.length > 0) {
          const tripIds = semanticTrips.map((t) => t.id);
          const activeTickets = await prisma.tickets.findMany({
            where: {
              tripId: { in: tripIds },
              status: { in: ["HELD", "PENDING", "CONFIRMED"] },
            },
            select: { tripId: true, seatNumber: true },
          });

          semanticTrips.forEach((trip) => {
            if (!resultMap.has(trip.id)) {
              const tripTickets = activeTickets.filter(
                (t) => t.tripId === trip.id,
              );
              const bookedSeatNumbers = tripTickets.map((t) => t.seatNumber);
              const totalSeats = trip.totalSeats ?? 30;

              resultMap.set(trip.id, {
                ...trip,
                totalSeats,
                availableSeats: Math.max(
                  0,
                  totalSeats - bookedSeatNumbers.length,
                ),
                bookedSeatNumbers,
              });
            }
          });
        }
      } catch (aiError: any) {
        console.warn(
          "[AI SEARCH] Lỗi gọi AI, chỉ trả về kết quả Text:",
          aiError.message,
        );
      }
    }

    return Array.from(resultMap.values()).slice(0, limit);
  }

import { GoogleGenAI } from "@google/genai";
import { GoogleGenerativeAI } from "@google/generative-ai";
import CircuitBreaker from "opossum";

interface AIProvider {
  name: string;
  apiKey: string;
  modelName: string;
}

export class AIProxyService {
  private static providers: AIProvider[] = [
    {
      name: "Gemini-Primary",
      apiKey: process.env.GEMINI_PRIMARY_KEY || "",
      modelName: "gemini-3.6-flash",
    },
    {
      name: "Gemini-Secondary",
      apiKey: process.env.GEMINI_SECONDARY_KEY || "",
      modelName: "gemini-3.6-flash",
    },
  ];

  // lưu trữ các Circuit Breaker trong bộ nhớ để tái sử dụng
  private static breakers: Map<string, CircuitBreaker> = new Map();

  private static getBreaker(provider: AIProvider): CircuitBreaker {
    if (!this.breakers.has(provider.name)) {
      // Hàm gọi API thực tế (Action)
      const action = async (prompt: string) => {
        const genAI = new GoogleGenerativeAI(provider.apiKey);
        const model = genAI.getGenerativeModel({ model: provider.modelName });
        const response = await model.generateContent(prompt);
        return response.response.text();
      };

      // Khởi tạo Cầu dao (Circuit Breaker)
      const breaker = new CircuitBreaker(action, {
        timeout: 20000, // Thời gian chờ tối đa 1 request là 20s
        errorThresholdPercentage: 50, // Ngắt mạch nếu > 50% request bị lỗi
        volumeThreshold: 5, // Phải có ít nhất 5 request bị lỗi thì mới xét ngắt mạch
        resetTimeout: 60000, // Thử mở lại mạch sau 1 phút
      });

      // Khi mạch bị ngắt (Open), lập tức ném lỗi Fallback mà không gọi API
      breaker.fallback(() => {
        throw new Error(
          `[Circuit Open] ${provider.name} đang tạm ngưng do lỗi liên tục.`,
        );
      });

      // Log trạng thái để dễ debug
      breaker.on("open", () =>
        console.warn(`[Circuit Breaker] 🔴 MẠCH ĐÃ NGẮT cho ${provider.name}`),
      );
      breaker.on("halfOpen", () =>
        console.log(`[Circuit Breaker] 🟡 ĐANG THỬ LẠI cho ${provider.name}`),
      );
      breaker.on("close", () =>
        console.log(
          `[Circuit Breaker] 🟢 MẠCH ĐÃ ĐÓNG LẠI cho ${provider.name}`,
        ),
      );

      this.breakers.set(provider.name, breaker);
    }
    return this.breakers.get(provider.name)!;
  }

  static async generateWithFallback(prompt: string): Promise<string> {
    let lastError: any = null;

    for (const provider of this.providers) {
      if (!provider.apiKey) continue;

      try {
        console.log(`[AI Proxy] Đang xử lý qua Provider: ${provider.name}...`);
        const breaker = this.getBreaker(provider);

        // Bắn request qua Cầu dao (Cầu dao sẽ tự xử lý Timeout)
        const responseText = await breaker.fire(prompt);

        if (responseText) {
          return responseText as any;
        }
      } catch (error: any) {
        lastError = error;
        console.warn(
          `[AI Proxy] Provider ${provider.name} thất bại (${error.message}). Đang tự động chuyển sang Provider dự phòng...`,
        );
      }
    }

    throw new Error(
      `[AI Proxy Critical] Tất cả các AI Provider/Key dự phòng đều sập! Error: ${lastError?.message}`,
    );
  }

  static async getEmbeddingWithFallback(text: string): Promise<number[]> {
    const textInput = String(text).trim();
    if (!textInput) {
      throw new Error("Text input for embedding cannot be empty");
    }

    let lastError: any = null;

    for (const provider of this.providers) {
      if (!provider.apiKey) continue;

      try {
        console.log(
          `[AI Proxy Embedding] Đang tạo Vector qua: ${provider.name}...`,
        );
        const ai = new GoogleGenAI({ apiKey: provider.apiKey });

        const response = await ai.models.embedContent({
          model: "gemini-embedding-2",
          contents: [
            {
              role: "user",
              parts: [{ text: textInput }],
            },
          ],
          config: {
            outputDimensionality: 1536,
          },
        });

        const embeddingValues = response.embeddings?.[0]?.values ?? [];
        if (embeddingValues.length > 0) {
          return embeddingValues;
        }
      } catch (error: any) {
        lastError = error;
        console.warn(
          `[AI Proxy Embedding] ${provider.name} gặp sự cố (${error.message}). Đang chuyển sang ${provider.name === "Gemini-Primary" ? "GEMINI_SECONDARY_KEY" : "kết thúc"}...`,
        );
      }
    }

    throw new Error(
      `[AI Proxy Critical] Không thể tạo Embedding từ cả Primary lẫn Secondary Key! Error: ${lastError?.message}`,
    );
  }
}

import prisma from "../database/prismaClient";
import { AIReconciliationAnalyzer } from "./aiReconciliationAnalyzer.service";
import { sendNotificationEmail } from "../queues/mailQueue";
import { AuditSigner } from "../Utils/auditSigner";

export interface MismatchDetail {
  userId: number;
  walletId: number;
  expectedBalance: number;
  actualBalance: number;
  difference: number;
  status: "DEBITED_AND_FROZEN" | "DEBT_GENERATED_AND_FROZEN";
}

const ACTION_SIGN: Record<string, 1 | -1> = {
  TOP_UP: 1,
  MASS_GIFT_RECEIVED: 1,
  MASS_GIFT: 1,
  REFUND_TICKET_PAYMENT: 1,
  CANCEL_TICKET_REFUND: 1,
  CREDIT: 1,

  BOOK_TICKET_PAYMENT: -1,
  BOOK_TICKET_BATCH: -1,
  MASS_GIFT_REVOKED: -1,
  REVOKE_BATCH: -1,
  WITHDRAW: -1,
  DEBIT: -1,
  SYSTEM_AUTO_DEBIT_AND_FREEZE: -1,
};

export class ReconciliationService {
  static async runDailyReconciliation() {
    console.log(
      "[RECONCILIATION] Bắt đầu tiến trình đối soát & Tự động phục hồi...",
    );
    const startTime = Date.now();

    const reconBatchId = `AIOPS_RECON_${new Date().toISOString().split("T")[0]}`;

    const mismatchedAccounts: MismatchDetail[] = [];
    let totalWalletsChecked = 0;
    let cursorId: number | undefined = undefined;
    const BATCH_SIZE = 100;

    try {
      while (true) {
        const wallets = await prisma.wallets.findMany({
          take: BATCH_SIZE,
          skip: cursorId ? 1 : 0,
          cursor: cursorId ? { id: cursorId } : undefined,
          orderBy: { id: "asc" },
          include: { user: { select: { email: true, fullName: true } } },
        });

        if (wallets.length === 0) break;
        cursorId = wallets[wallets.length - 1].id;
        totalWalletsChecked += wallets.length;

        const userIds = wallets.map((w) => w.userId);

        const allAuditLogs = await prisma.auditLog.findMany({
          where: {
            userId: { in: userIds },
            resource: "Wallets",
            action: {
              notIn: [
                "SYSTEM_ALERT_BALANCE_MISMATCH",
                "SYSTEM_AUTO_DEBIT_AND_FREEZE",
                "ADMIN_UNLOCK_WALLET",
              ],
            },
          },
          orderBy: { createdAt: "asc" },
        });

        const logsByUser = new Map<number, typeof allAuditLogs>();
        for (const log of allAuditLogs) {
          if (!log.userId) continue;
          if (!logsByUser.has(log.userId)) logsByUser.set(log.userId, []);
          logsByUser.get(log.userId)!.push(log);
        }

        for (const wallet of wallets) {
          const auditLogs = logsByUser.get(wallet.userId) || [];
          let calculatedBalance = 0;
          let hasTamperedLog = false;

          for (const log of auditLogs) {
            const newData = log.newData as Record<string, any> | null;
            if (!newData || typeof newData.amount !== "number") continue;

            const isValidSignature = AuditSigner.verifySignature(
              {
                requestId: log.requestId,
                userId: log.userId,
                action: log.action,
                resource: log.resource,
                resourceId: log.resourceId,
                newData: log.newData,
                oldData: log.oldData,
              },
              log.signature || "",
            );

            if (!isValidSignature) {
              console.error(
                `[SECURITY ALERT] Phát hiện AuditLog ID ${log.id} của User ${wallet.userId} bị can thiệp trái phép!`,
              );
              hasTamperedLog = true;
              continue;
            }

            const rawAmount = newData.amount;
            const action = log.action.toUpperCase();
            const sign = ACTION_SIGN[action];

            if (sign === 1) {
              calculatedBalance += Math.abs(rawAmount);
            } else if (sign === -1) {
              calculatedBalance -= Math.abs(rawAmount);
            } else {
              calculatedBalance += rawAmount;
            }
          }

          const actualBalance = Number(wallet.balance);
          const difference = Number(
            (actualBalance - calculatedBalance).toFixed(2),
          );

          if (Math.abs(difference) > 0.001 || hasTamperedLog) {
            const isDebt = calculatedBalance < 0;

            mismatchedAccounts.push({
              userId: wallet.userId,
              walletId: wallet.id,
              expectedBalance: calculatedBalance,
              actualBalance: actualBalance,
              difference: difference,
              status: hasTamperedLog
                ? "DEBITED_AND_FROZEN"
                : isDebt
                  ? "DEBT_GENERATED_AND_FROZEN"
                  : "DEBITED_AND_FROZEN",
            });

            try {
              await prisma.$transaction(async (tx) => {
                await tx.wallets.update({
                  where: { id: wallet.id },
                  data: {
                    balance: hasTamperedLog ? actualBalance : calculatedBalance,
                    isLocked: true,
                  },
                });

                await tx.auditLog.create({
                  data: {
                    requestId: `AIOPS-SECURITY-${Date.now()}-${wallet.id}`,
                    userId: wallet.userId,
                    action: "SYSTEM_AUTO_DEBIT_AND_FREEZE",
                    batchId: reconBatchId,
                    resource: "Wallets",
                    resourceId: wallet.id.toString(),
                    oldData: { actualBalance, isLocked: wallet.isLocked },
                    newData: {
                      amount: hasTamperedLog ? 0 : -difference,
                      adjustedAmount: hasTamperedLog ? 0 : -difference,
                      newBalance: hasTamperedLog
                        ? actualBalance
                        : calculatedBalance,
                      isLocked: true,
                      isDebt: isDebt,
                      reason: hasTamperedLog
                        ? "Phát hiện can thiệp trái phép vào lịch sử giao dịch (AuditLog bị sửa) -> Khóa ví bảo vệ, không tự động trừ tiền!"
                        : "Cân bằng lại số dư chuẩn & Khóa ví chờ Admin duyệt",
                    },
                    ipAddress: "127.0.0.1 (CRON_AIOPS)",
                  },
                });
              });

              if (wallet.user?.email) {
                await sendNotificationEmail({
                  email: wallet.user.email,
                  type: "WALLET_LOCKED_AI",
                  data: {
                    userName: wallet.user.fullName,
                    actualBalance: actualBalance,
                    calculatedBalance: calculatedBalance,
                  },
                }).catch((err) =>
                  console.error(
                    "[RECONCILIATION] Lỗi đẩy job gửi mail khóa ví:",
                    err,
                  ),
                );
              }
            } catch (txError: any) {
              console.error(
                `[RECONCILIATION] Lỗi khi xử lý ví ${wallet.id}:`,
                txError.message,
              );
            }
          }
        }
      }
    } catch (error: any) {
      console.error("[RECONCILIATION] Lỗi quét ví:", error.message);
    }

    const duration = Date.now() - startTime;

    let logsSample: any[] = [];
    let aiAnalysis = null;

    if (mismatchedAccounts.length > 0) {
      logsSample = await prisma.auditLog.findMany({
        where: { userId: { in: mismatchedAccounts.map((a) => a.userId) } },
        take: 5,
        orderBy: { createdAt: "desc" },
      });

      //Khai báo biến timeoutId để có thể clear
      let timeoutId: NodeJS.Timeout;

      try {
        const aiPromise = AIReconciliationAnalyzer.analyzeMismatches(
          mismatchedAccounts,
          totalWalletsChecked,
          logsSample,
        );

        const timeoutPromise = new Promise((_, reject) => {
          timeoutId = setTimeout(() => reject(new Error("AI_TIMEOUT")), 25000);
        });

        aiAnalysis = (await Promise.race([aiPromise, timeoutPromise])) as any;
      } catch (aiError: any) {
        console.error(
          "[RECONCILIATION] Lỗi khi gọi AI phân tích:",
          aiError.message,
        );
        aiAnalysis = {
          riskLevel: "HIGH",
          rootCauseAnalysis:
            "Hệ thống AI phản hồi quá chậm hoặc đang quá tải. Vui lòng thử lại sau.",
          recommendedAction:
            "Kiểm tra thủ công danh sách mảng mismatchedAccounts.",
        };
      } finally {
        if (timeoutId!) clearTimeout(timeoutId);
      }
    }

    return {
      success: true,
      totalWalletsChecked,
      mismatchCount: mismatchedAccounts.length,
      mismatchedAccounts,
      durationMs: duration,
      aiAnalysis,
    };
  }
}

      JOIN "Trips" tr ON t."tripId" = tr.id

        total_revenue as amount
      FROM revenue_stats
      WHERE date >= CURRENT_DATE - INTERVAL '7 days'
    `;

      }

        amount,

    );

            newBalance: newBalance,
            reason: data.reason,
          },
          ipAddress: null,
          isRevoked: false,

          requestId,
          userId: data.adminUserId,

        amountPerUser: data.amount,

  static async holdOrBookSeat(
    userId: number,
    tripId: number,
    seatNumber: number,
    paymentMethod: "WALLET" | "LATER" = "WALLET",
    batchId?: string,
  ) {
    const t0 = Date.now();
    const lockKey = `lock:trip:${tripId}:seat:${seatNumber}`;
    const lockToken = await RedisLock.acquire(lockKey, 30);
    const t1 = Date.now();

    if (!lockToken) {
      const error: any = new Error(
        `Ghế số ${seatNumber} đang có người khác thao tác chọn.`,
      );
      error.statusCode = 409;
      throw error;
    }

    let result: any;
    try {
      result = await prisma.$transaction(
        async (tx) => {
          await tx.$queryRaw`SELECT pg_advisory_xact_lock(hashtext(${`seat:${tripId}:${seatNumber}`}))::text`;
          const existingTicket = await tx.tickets.findFirst({
            where: {
              tripId,
              seatNumber,
              status: { in: ["HELD", "PENDING", "CONFIRMED"] },
            },
          });
          if (existingTicket) {
            const error: any = new Error(
              `Ghế số ${seatNumber} đã có người đặt hoặc đang giữ chỗ!`,
            );
            error.statusCode = 400;
            throw error;
          }
          const trip = await tx.trips.findUnique({ where: { id: tripId } });
          if (!trip) {
            const error: any = new Error("Chuyến xe không tồn tại!");
            error.statusCode = 404;
            throw error;
          }
          let initialStatus: "HELD" | "PENDING" | "CONFIRMED" = "PENDING";
          let currentBalance = 0;
          const tripPrice = Number(trip.price);
          let walletId = userId;

          if (paymentMethod === "WALLET") {
            const wallet = await tx.wallets.findUnique({ where: { userId } });
            if (!wallet || wallet.isLocked) {
              const error: any = new Error(
                "Ví của bạn đang bị khóa do vi phạm hoặc chờ đối soát.",
              );
              error.statusCode = 403;
              throw error;
            }
            currentBalance = Number(wallet.balance || 0);
            walletId = wallet.id;
            if (currentBalance < tripPrice) {
              const error: any = new Error(
                "Số dư ví không đủ để thanh toán vé này!",
              );
              error.statusCode = 400;
              throw error;
            }
            await tx.wallets.update({
              where: { userId },
              data: { balance: { decrement: tripPrice } },
            });
            initialStatus = "CONFIRMED";
          }

          const newTicket = await tx.tickets.create({
            data: {
              userId,
              tripId,
              seatNumber,
              status: initialStatus,
              paymentMethod: paymentMethod === "WALLET" ? "WALLET" : "LATER",
              paidAmount: paymentMethod === "WALLET" ? tripPrice : 0,
              ...(batchId ? { batchId } : {}),
            } as any,
            include: { user: { select: { email: true } } },
          });

          if (paymentMethod === "WALLET") {
            await tx.auditLog.create({
              data: {
                requestId: `BOOK_TICKET_${Date.now()}`,
                batchId: batchId || `TRIP_${tripId}`,
                userId,
                action: "BOOK_TICKET_PAYMENT",
                resource: "Wallets",
                resourceId: String(walletId),
                oldData: { balance: currentBalance },
                newData: {
                  amount: -tripPrice,
                  description: `Thanh toán vé chuyến ${trip.route} (Ghế A${seatNumber})`,
                  newBalance: currentBalance - tripPrice,
                  tripId,
                  seats: [seatNumber],
                  ticketId: newTicket.id,
                },
              },
            });
          }
          return newTicket;
        },
        { timeout: 15000 },
      );
    } finally {
      await RedisLock.release(lockKey, lockToken);
    }
    const t2 = Date.now();

    if (paymentMethod === "WALLET" && result?.id && result?.user?.email) {
      try {
        const effectiveBatchId = batchId || `SINGLE_${result.id}`;
        await scheduleNotificationWithReset(
          {
            email: result.user.email,
            type: "BOOKING_SUCCESS_BATCH",
            data: { userId, batchId: effectiveBatchId },
          },
          { delayMs: 10000, jobId: `booking-notif-${effectiveBatchId}` },
        );
      } catch (mailErr: any) {
        console.error("[BOOKING] Schedule mail error:", mailErr?.message);
      }
    }
    const t3 = Date.now();

    void Promise.all([
      MultiLevelCache.invalidate(`trips:detail:${tripId}`),
      MultiLevelCache.invalidate("trips:all:active"),
    ]).catch((err) =>
      console.warn("[BOOKING] cache invalidation:", err?.message),
    );

    console.log(
      `[TIMING][BOOK] lock=${t1 - t0}ms tx=${t2 - t1}ms mail=${t3 - t2}ms total=${t3 - t0}ms`,
    );
    return result;
  }

  static async bookMultipleSeats(
    userId: number,
    tripId: number,
    seatNumbers: number[],
    paymentMethod: "WALLET" | "LATER" = "WALLET",
    batchId?: string,
  ) {
    const t0 = Date.now();
    if (!seatNumbers || seatNumbers.length === 0) {
      const error: any = new Error("Vui lòng chọn ít nhất 1 ghế!");
      error.statusCode = 400;
      throw error;
    }

    const sortedSeats = [...seatNumbers].sort((a, b) => a - b);
    const lockKey = `lock:trip:${tripId}`;
    const lockToken = await RedisLock.acquire(lockKey, 30);
    const t1 = Date.now();

    if (!lockToken) {
      const error: any = new Error("Hệ thống đang bận, vui lòng thử lại sau!");
      error.statusCode = 409;
      throw error;
    }

    let resultTickets: any[] = [];
    const effectiveBatchId = batchId || `BATCH_${Date.now()}`;

    try {
      resultTickets = await prisma.$transaction(
        async (tx) => {
          await tx.$queryRaw`SELECT pg_advisory_xact_lock(hashtext(${`trip:${tripId}`}))::text`;
          const existingTickets = await tx.tickets.findMany({
            where: {
              tripId,
              seatNumber: { in: sortedSeats },
              status: { in: ["HELD", "PENDING", "CONFIRMED"] },
            },
          });
          if (existingTickets.length > 0) {
            const takenSeats = existingTickets
              .map((t) => t.seatNumber)
              .join(", ");
            const error: any = new Error(
              `Các ghế sau đã có người đặt: ${takenSeats}`,
            );
            error.statusCode = 400;
            throw error;
          }

          const trip = await tx.trips.findUnique({ where: { id: tripId } });
          if (!trip) {
            const error: any = new Error("Chuyến xe không tồn tại!");
            error.statusCode = 404;
            throw error;
          }

          const tripPrice = Number(trip.price);
          const totalPrice = tripPrice * sortedSeats.length;
          let currentBalance = 0;
          let walletId = userId;
          let initialStatus: "PENDING" | "CONFIRMED" = "PENDING";

          if (paymentMethod === "WALLET") {
            const wallet = await tx.wallets.findUnique({ where: { userId } });
            if (!wallet || wallet.isLocked) {
              const error: any = new Error(
                "Ví của bạn đang bị khóa do vi phạm hoặc chờ đối soát.",
              );
              error.statusCode = 403;
              throw error;
            }
            currentBalance = Number(wallet.balance || 0);
            walletId = wallet.id;
            if (currentBalance < totalPrice) {
              const error: any = new Error(
                "Số dư ví không đủ để thanh toán các vé này!",
              );
              error.statusCode = 400;
              throw error;
            }
            await tx.wallets.update({
              where: { userId },
              data: { balance: { decrement: totalPrice } },
            });
            initialStatus = "CONFIRMED";
          }

          const ticketData = sortedSeats.map((seat) => ({
            userId,
            tripId,
            seatNumber: seat,
            status: initialStatus,
            paymentMethod: paymentMethod === "WALLET" ? "WALLET" : "LATER",
            paidAmount: paymentMethod === "WALLET" ? tripPrice : 0,
            batchId: effectiveBatchId,
          }));

          await tx.tickets.createMany({ data: ticketData as any });

          const newTickets = await tx.tickets.findMany({
            where: { batchId: effectiveBatchId, userId, tripId },
            include: { user: { select: { email: true } } },
          });

          if (paymentMethod === "WALLET") {
            await tx.auditLog.create({
              data: {
                requestId: `BOOK_TICKET_BATCH_${Date.now()}`,
                batchId: effectiveBatchId,
                userId,
                action: "BOOK_TICKET_PAYMENT",
                resource: "Wallets",
                resourceId: String(walletId),
                oldData: { balance: currentBalance },
                newData: {
                  amount: -totalPrice,
                  description: `Thanh toán ${sortedSeats.length} vé chuyến ${trip.route} (Ghế: ${sortedSeats.join(", ")})`,
                  newBalance: currentBalance - totalPrice,
                  tripId,
                  seats: sortedSeats,
                  ticketIds: newTickets.map((t) => t.id),
                },
              },
            });
          }
          return newTickets;
        },
        { timeout: 20000 },
      );
    } finally {
      await RedisLock.release(lockKey, lockToken);
    }
    const t2 = Date.now();

    if (
      paymentMethod === "WALLET" &&
      resultTickets.length > 0 &&
      resultTickets[0].user?.email
    ) {
      try {
        const email = resultTickets[0].user.email;
        await scheduleNotificationWithReset(
          {
            email,
            type: "BOOKING_SUCCESS_BATCH",
            data: { userId, batchId: effectiveBatchId },
          },
          { delayMs: 10000, jobId: `booking-notif-${effectiveBatchId}` },
        );
      } catch (mailErr: any) {
        console.error("[BOOKING] Schedule mail error:", mailErr?.message);
      }
    }
    const t3 = Date.now();

    void Promise.all([
      MultiLevelCache.invalidate(`trips:detail:${tripId}`),
      MultiLevelCache.invalidate("trips:all:active"),
    ]).catch((err) =>
      console.warn("[BOOKING] cache invalidation:", err?.message),
    );

    console.log(
      `[TIMING][BULK_BOOK] lock=${t1 - t0}ms tx=${t2 - t1}ms mail=${t3 - t2}ms total=${t3 - t0}ms`,
    );
    return resultTickets;
  }

  static async adminRevokeTicket(
    ticketId: number,
    adminUserId: number,
    reason: string = "Admin thu hồi thủ công",
  ) {
    const ticket = await prisma.tickets.findUnique({
      where: { id: ticketId },
      include: { trip: true, user: true },
    });
    if (!ticket) throw new Error("Không tìm thấy vé!");
    if (ticket.status === "REVOKED_BY_ADMIN" || ticket.status === "CANCELED")
      throw new Error("Vé này đã bị thu hồi/hủy trước đó!");

    const result = await prisma.$transaction(async (tx) => {
      const updated = await tx.tickets.update({
        where: { id: ticketId },
        data: { status: "REVOKED_BY_ADMIN" },
      });
      let refundAmount = 0;
      if (ticket.status === "CONFIRMED")
        refundAmount = Number(ticket.paidAmount ?? ticket.trip.price ?? 0);

      if (refundAmount > 0) {
        const wallet = await tx.wallets.findUnique({
          where: { userId: ticket.userId },
        });
        if (!wallet)
          throw new Error(
            "Không tìm thấy ví để hoàn tiền! Dữ liệu vé bất thường.",
          );
        const currentBalance = Number(wallet.balance || 0);

        await tx.wallets.update({
          where: { userId: ticket.userId },
          data: { balance: { increment: refundAmount } },
        });
        await tx.auditLog.create({
          data: {
            requestId: `ADMIN_REVOKE_REFUND_${Date.now()}`,
            userId: ticket.userId,
            action: "REFUND_TICKET_PAYMENT",
            batchId: `ADMIN_REVOKE_${ticketId}`,
            resource: "Wallets",
            resourceId: String(wallet.id),
            oldData: { balance: currentBalance },
            newData: {
              amount: refundAmount,
              description: `Hoàn tiền vé bị thu hồi (Chuyến ${ticket.trip.route}, Ghế A${ticket.seatNumber}). Lý do: ${reason}`,
              newBalance: currentBalance + refundAmount,
            },
          },
        });
      }

      await tx.auditLog.create({
        data: {
          requestId: `ADMIN_REVOKE_TICKET_${Date.now()}`,
          userId: ticket.userId,
          action: "ADMIN_REVOKE_TICKET",
          resource: "Tickets",
          resourceId: String(ticketId),
          oldData: { status: ticket.status },
          newData: {
            status: "REVOKED_BY_ADMIN",
            reason,
            refunded: refundAmount,
            performedByAdminId: adminUserId,
          },
        },
      });
      return updated;
    });

    if (ticket.user) {
      try {
        const effectiveBatchId =
          (ticket as any).batchId || `SINGLE_${ticket.id}`;
        await scheduleNotificationWithReset(
          {
            email: ticket.user.email,
            type: "ADMIN_REVOKE_BATCH",
            data: { userId: ticket.userId, batchId: effectiveBatchId, reason },
          },
          { delayMs: 30000, jobId: `revoke-notif-${effectiveBatchId}` },
        );
      } catch (mailErr: any) {
        console.error("[BOOKING] Schedule revoke mail:", mailErr?.message);
      }
    }

    void Promise.all([
      MultiLevelCache.invalidate(`trips:detail:${ticket.tripId}`),
      MultiLevelCache.invalidate("trips:all:active"),
      MultiLevelCache.invalidate(`user:${ticket.userId}:tickets*`),
    ]).catch((err) =>
      console.warn("[BOOKING] cache invalidation:", err?.message),
    );

    return result;
  }

  static async cancelTicket(userId: number, ticketId: number) {
    const ticket = await prisma.tickets.findUnique({
      where: { id: ticketId },
      include: { trip: true, user: true },
    });
    if (!ticket) {
      const error: any = new Error("Không tìm thấy vé đặt!");
      error.statusCode = 404;
      throw error;
    }
    if (ticket.userId !== userId) {
      const error: any = new Error("Bạn không có quyền hủy vé này!");
      error.statusCode = 403;
      throw error;
    }
    if (
      ticket.status === "CANCELED" ||
      ticket.status === "USED" ||
      ticket.status === "REVOKED_BY_ADMIN"
    ) {
      const error: any = new Error("Vé này không thể hủy được nữa!");
      error.statusCode = 400;
      throw error;
    }

    const result = await prisma.$transaction(async (tx) => {
      const updatedTicket = await tx.tickets.update({
        where: { id: ticketId },
        data: { status: "CANCELED" },
      });

      if (ticket.status === "CONFIRMED") {
        const refundAmount = Number(
          ticket.paidAmount ?? ticket.trip.price ?? 0,
        );
        if (refundAmount > 0) {
          const wallet = await tx.wallets.findUnique({ where: { userId } });
          if (!wallet) throw new Error("Không tìm thấy ví để hoàn tiền!");
          const currentBalance = Number(wallet.balance || 0);

          await tx.wallets.update({
            where: { userId },
            data: { balance: { increment: refundAmount } },
          });
          await tx.auditLog.create({
            data: {
              requestId: `REFUND_TICKET_${Date.now()}`,
              userId,
              action: "REFUND_TICKET_PAYMENT",
              resource: "Wallets",
              resourceId: String(wallet.id),
              oldData: { balance: currentBalance },
              newData: {
                amount: refundAmount,
                description: `Hoàn tiền hủy vé chuyến ${ticket.trip.route} (Ghế A${ticket.seatNumber})`,
                newBalance: currentBalance + refundAmount,
              },
            },
          });
        }
      }
      return updatedTicket;
    });

    if (ticket.user) {
      try {
        await sendNotificationEmail({
          email: ticket.user.email,
          type: "BOOKING_CANCEL",
          data: {
            route: ticket.trip.route,
            seat: ticket.seatNumber,
          },
        });
      } catch (mailErr: any) {
        console.error("[BOOKING] Hủy vé mail error:", mailErr?.message);
      }
    }

    void Promise.all([
      MultiLevelCache.invalidate(`trips:detail:${ticket.tripId}`),
      MultiLevelCache.invalidate("trips:all:active"),
      MultiLevelCache.invalidate(`user:${userId}:tickets*`),
    ]).catch((err) =>
      console.warn("[BOOKING] cache invalidation:", err?.message),
    );

    return result;
  }

  static async adminRevokeMultipleTickets(
    ticketIds: number[],
    adminUserId: number,
    reason: string = "Admin thu hồi nhiều vé",
  ) {
    if (!ticketIds || ticketIds.length === 0)
      throw new Error("Chưa chọn vé nào!");

    const tickets = await prisma.tickets.findMany({
      where: { id: { in: ticketIds }, status: "CONFIRMED" },
      include: { user: true, trip: true },
    });

    if (tickets.length === 0) throw new Error("Không có vé hợp lệ để thu hồi!");

    const refundByUser = new Map<number, number>();
    tickets.forEach((t) => {
      const amount = Number(t.paidAmount ?? t.trip.price ?? 0);
      if (amount > 0) {
        refundByUser.set(t.userId, (refundByUser.get(t.userId) || 0) + amount);
      }
    });

    await prisma.$transaction(async (tx) => {
      await tx.tickets.updateMany({
        where: { id: { in: tickets.map((t) => t.id) } },
        data: { status: "REVOKED_BY_ADMIN" },
      });

      for (const [userId, amount] of refundByUser.entries()) {
        const wallet = await tx.wallets.findUnique({ where: { userId } });
        if (wallet) {
          const currentBalance = Number(wallet.balance);
          await tx.wallets.update({
            where: { userId },
            data: { balance: { increment: amount } },
          });
          await tx.auditLog.create({
            data: {
              requestId: `BULK_REVOKE_REFUND_${Date.now()}_${userId}`,
              userId: userId,
              action: "REFUND_TICKET_PAYMENT",
              batchId: `BULK_REVOKE_${Date.now()}`,
              resource: "Wallets",
              resourceId: String(wallet.id),
              oldData: { balance: currentBalance },
              newData: {
                amount: amount,
                description: `Hoàn tiền ${tickets.filter((t) => t.userId === userId).length} vé bị thu hồi. Lý do: ${reason}`,
                newBalance: currentBalance + amount,
              },
            },
          });
        }
      }

      await tx.auditLog.create({
        data: {
          requestId: `BULK_REVOKE_${Date.now()}`,
          userId: adminUserId,
          action: "ADMIN_REVOKE_MULTIPLE_TICKETS",
          resource: "Tickets",
          resourceId: ticketIds.join(","),
          newData: {
            ticketIds: tickets.map((t) => t.id),
            reason,
            performedByAdminId: adminUserId,
            totalTickets: tickets.length,
          },
        },
      });
    });

    const ticketsByUser = new Map<number, typeof tickets>();
    tickets.forEach((t) => {
      const arr = ticketsByUser.get(t.userId) || [];
      arr.push(t);
      ticketsByUser.set(t.userId, arr);
    });

    for (const [userId, userTickets] of ticketsByUser.entries()) {
      const email = userTickets[0].user?.email;
      if (email) {
        try {
          await scheduleNotificationWithReset(
            {
              email,
              type: "ADMIN_REVOKE_MULTIPLE",
              data: {
                userId,
                ticketIds: userTickets.map((t) => t.id),
                reason,
              },
            },
            {
              delayMs: 10000,
              jobId: `revoke-multi-${userId}-${Date.now()}`,
            },
          );
        } catch (err) {
          console.error("Lỗi gửi mail thu hồi nhiều vé:", err);
        }
      }
    }

    return { success: true, count: tickets.length };
  }
